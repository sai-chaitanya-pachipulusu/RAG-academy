# Self-RAG: Retrieval Necessity Predictor

Not every question needs retrieval. "What is 2+2?" doesn't require searching your vector database.

## The Self-RAG Framework

Before retrieving, ask: **"Do I need external knowledge?"**

| Query Type                          | Decision                    |
| ----------------------------------- | --------------------------- |
| "What is the capital of France?"    | Generate (common knowledge) |
| "What was Apple's Q3 2024 revenue?" | Retrieve (specific data)    |
| "What does our refund policy say?"  | Retrieve (private data)     |
| "Tell me a joke"                    | Generate (no facts needed)  |

## Decision Framework

```
Query
  │
  ├── Is it harmful/off-topic? ──→ REFUSE
  │
  ├── Is it common knowledge? ───→ GENERATE
  │
  ├── Is it private/internal? ───→ RETRIEVE
  │
  ├── Is it time-sensitive? ─────→ RETRIEVE
  │
  └── Uncertain? ────────────────→ RETRIEVE (safe default)
```

## Implementation

### Rule-Based Classifier

```python
class RuleBasedGrader:
    """Simple rule-based retrieval necessity classifier."""

    def __init__(self):
        # Common knowledge patterns
        self.common_knowledge_patterns = [
            r"what is \d+\s*\+\s*\d+",  # Basic math
            r"capital of [a-z]+",         # Geography
            r"who (is|was) [a-z]+\s*$",   # Famous people
        ]

        # Private data indicators
        self.private_indicators = [
            "policy", "refund", "return", "warranty",
            "internal", "company", "our", "we"
        ]

        # Time-sensitive patterns
        self.time_patterns = [
            r"\b(2024|2025|today|yesterday|last week|this month)\b",
            r"current", "latest", "recent", "now"
        ]

    def grade(self, query: str) -> str:
        """
        Grade query for retrieval necessity.

        Returns: "generate", "retrieve", or "refuse"
        """
        query_lower = query.lower()

        # Check for harmful content (simplified)
        if self._is_harmful(query_lower):
            return "refuse"

        # Check for common knowledge
        if self._is_common_knowledge(query):
            return "generate"

        # Check for private/internal data
        if self._is_private(query_lower):
            return "retrieve"

        # Check for time-sensitivity
        if self._is_time_sensitive(query):
            return "retrieve"

        # Default to retrieve for safety
        return "retrieve"

    def _is_harmful(self, query: str) -> bool:
        """Check for harmful/off-topic queries."""
        harmful_terms = ["hack", "steal", "password", "credit card"]
        return any(term in query for term in harmful_terms)

    def _is_common_knowledge(self, query: str) -> bool:
        """Check if query asks for common knowledge."""
        import re
        for pattern in self.common_knowledge_patterns:
            if re.search(pattern, query, re.IGNORECASE):
                return True
        return False

    def _is_private(self, query: str) -> bool:
        """Check if query asks for private/company data."""
        return any(indicator in query for indicator in self.private_indicators)

    def _is_time_sensitive(self, query: str) -> bool:
        """Check if query asks for time-sensitive information."""
        import re
        for pattern in self.time_patterns:
            if re.search(pattern, query, re.IGNORECASE):
                return True
        return False
```

### LLM-Based Classifier

```python
class LLMGrader:
    """LLM-based retrieval necessity classifier."""

    GRADING_PROMPT = """You are a classifier that decides how to handle user queries.

Query: {query}

Classify into one of these categories:

1. GENERATE - Use parametric knowledge (common facts, math, creative tasks)
   Examples: "What is 2+2?", "Capital of France?", "Write a poem"

2. RETRIEVE - Search external documents (private data, recent info, specific facts)
   Examples: "What was Q3 revenue?", "Company refund policy?", "Latest news"

3. REFUSE - Reject harmful or off-topic queries
   Examples: "How to hack?", "Give me passwords", "Illegal activity"

Respond with exactly one word: GENERATE, RETRIEVE, or REFUSE

Classification:"""

    def __init__(self, llm_client):
        self.llm = llm_client

    def grade(self, query: str) -> str:
        """Grade query using LLM."""
        prompt = self.GRADING_PROMPT.format(query=query)
        response = self.llm.generate(prompt, max_tokens=10)

        # Parse response
        response_upper = response.strip().upper()

        if "REFUSE" in response_upper:
            return "refuse"
        elif "GENERATE" in response_upper:
            return "generate"
        else:
            return "retrieve"  # Safe default

    def grade_with_confidence(self, query: str) -> dict:
        """Grade with confidence score."""
        prompt = self.GRADING_PROMPT.format(query=query) + """

Also provide confidence (0.0-1.0) and brief reasoning.

Format: DECISION|CONFIDENCE|REASONING"""

        response = self.llm.generate(prompt, max_tokens=50)

        try:
            parts = response.strip().split("|")
            return {
                "decision": parts[0].lower().strip(),
                "confidence": float(parts[1]),
                "reasoning": parts[2] if len(parts) > 2 else ""
            }
        except:
            return {"decision": "retrieve", "confidence": 0.5, "reasoning": "parse_error"}
```

### Hybrid Classifier

```python
class HybridGrader:
    """Combine rule-based and LLM-based grading for speed and accuracy."""

    def __init__(self, llm_client, confidence_threshold: float = 0.8):
        self.rule_grader = RuleBasedGrader()
        self.llm_grader = LLMGrader(llm_client)
        self.confidence_threshold = confidence_threshold

    def grade(self, query: str) -> str:
        """
        Grade query using fast rules first, then LLM if uncertain.

        Optimization: 70% of queries resolved by rules in <1ms
        """
        # Try rule-based first (fast)
        rule_result = self.rule_grader.grade(query)

        # High-confidence rule decisions
        if rule_result == "refuse":
            return "refuse"  # Always refuse harmful

        if rule_result == "generate" and self._is_clear_common_knowledge(query):
            return "generate"  # Clear math, geography, etc.

        if rule_result == "retrieve" and self._is_clearly_private(query):
            return "retrieve"  # Clear internal query

        # Use LLM for edge cases
        llm_result = self.llm_grader.grade_with_confidence(query)

        if llm_result["confidence"] >= self.confidence_threshold:
            return llm_result["decision"]

        # Low confidence - default to retrieve (safer)
        return "retrieve"

    def _is_clear_common_knowledge(self, query: str) -> bool:
        """Check for unambiguous common knowledge queries."""
        # Math expressions
        import re
        if re.match(r"^\s*\d+\s*[+\-*/]\s*\d+\s*$", query):
            return True

        # Simple definitions
        if re.match(r"^what is (the )?[a-z]+\??$", query.lower()):
            return True

        return False

    def _is_clearly_private(self, query: str) -> bool:
        """Check for clearly private/internal queries."""
        query_lower = query.lower()
        private_markers = ["our company", "our policy", "internal", "employee"]
        return any(marker in query_lower for marker in private_markers)
```

## Integration with RAG Pipeline

```python
class SmartRAG:
    """RAG system with retrieval necessity prediction."""

    def __init__(self, retriever, llm, grader):
        self.retriever = retriever
        self.llm = llm
        self.grader = grader

    def query(self, question: str) -> dict:
        """Route query to appropriate handler."""
        # Grade the query
        decision = self.grader.grade(question)

        if decision == "refuse":
            return {
                "answer": "I cannot answer that question.",
                "source": "refusal",
                "reason": "harmful_or_inappropriate"
            }

        elif decision == "generate":
            # Generate without retrieval
            answer = self.llm.generate(question)
            return {
                "answer": answer,
                "source": "parametric_knowledge",
                "retrieval_skipped": True
            }

        else:  # retrieve
            # Full RAG pipeline
            docs = self.retriever.search(question)
            context = self._format_docs(docs)

            prompt = f"""Answer based on the context:

Context:
{context}

Question: {question}

Answer:"""

            answer = self.llm.generate(prompt)

            return {
                "answer": answer,
                "source": "retrieval",
                "documents": [d["id"] for d in docs],
                "retrieval_skipped": False
            }

    def _format_docs(self, docs: list) -> str:
        """Format documents for context."""
        return "\n\n".join([f"[Doc {i+1}] {d['text']}" for i, d in enumerate(docs)])
```

## Evaluation

```python
def evaluate_grader(grader, test_set: list):
    """Evaluate grader on labeled test set."""
    correct = 0
    total = len(test_set)

    for item in test_set:
        predicted = grader.grade(item["query"])
        if predicted == item["label"]:
            correct += 1

    accuracy = correct / total

    # Per-class metrics
    from collections import defaultdict
    per_class = defaultdict(lambda: {"tp": 0, "fp": 0, "fn": 0})

    for item in test_set:
        predicted = grader.grade(item["query"])
        actual = item["label"]

        if predicted == actual:
            per_class[actual]["tp"] += 1
        else:
            per_class[predicted]["fp"] += 1
            per_class[actual]["fn"] += 1

    return {
        "accuracy": accuracy,
        "per_class": dict(per_class)
    }
```
