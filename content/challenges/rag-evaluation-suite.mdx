## The Evaluation Challenge

RAG systems fail in subtle ways:

- Retrieval can miss relevant docs
- Context can be noisy
- Generation can hallucinate
- End-to-end quality is hard to measure

## Evaluation Framework

### Three-Level Evaluation

```
┌─────────────────────────────────────────┐
│      Level 3: End-to-End Quality        │
│   (Answer quality, user satisfaction)    │
├─────────────────────────────────────────┤
│       Level 2: Generation Quality        │
│   (Faithfulness, relevance, coherence)   │
├─────────────────────────────────────────┤
│       Level 1: Retrieval Quality         │
│     (Recall, precision, relevance)       │
└─────────────────────────────────────────┘
```

## Level 1: Retrieval Metrics

### Core Metrics

```python
import numpy as np

def recall_at_k(retrieved: list[str], relevant: list[str], k: int) -> float:
    """What fraction of relevant docs were retrieved?"""
    retrieved_set = set(retrieved[:k])
    relevant_set = set(relevant)

    if not relevant_set:
        return 1.0

    return len(retrieved_set & relevant_set) / len(relevant_set)

def precision_at_k(retrieved: list[str], relevant: list[str], k: int) -> float:
    """What fraction of retrieved docs are relevant?"""
    retrieved_set = set(retrieved[:k])
    relevant_set = set(relevant)

    if not retrieved_set:
        return 0.0

    return len(retrieved_set & relevant_set) / len(retrieved_set)

def mrr(retrieved: list[str], relevant: list[str]) -> float:
    """Mean Reciprocal Rank - where does first relevant doc appear?"""
    for i, doc_id in enumerate(retrieved):
        if doc_id in relevant:
            return 1 / (i + 1)
    return 0.0

def ndcg_at_k(retrieved: list[str], relevant: dict[str, float], k: int) -> float:
    """Normalized Discounted Cumulative Gain."""
    def dcg(scores):
        return sum(score / np.log2(i + 2) for i, score in enumerate(scores))

    # Actual DCG
    actual_scores = [relevant.get(doc_id, 0) for doc_id in retrieved[:k]]
    actual_dcg = dcg(actual_scores)

    # Ideal DCG
    ideal_scores = sorted(relevant.values(), reverse=True)[:k]
    ideal_dcg = dcg(ideal_scores)

    return actual_dcg / ideal_dcg if ideal_dcg > 0 else 0
```

## Level 2: Generation Metrics

### Faithfulness (Grounding)

```python
def faithfulness_score(response: str, contexts: list[str]) -> dict:
    """Is the response grounded in the provided context?"""
    # Extract claims from response
    claims = extract_claims(response)

    supported = 0
    unsupported = []

    for claim in claims:
        if is_supported_by_context(claim, contexts):
            supported += 1
        else:
            unsupported.append(claim)

    return {
        "score": supported / len(claims) if claims else 1.0,
        "total_claims": len(claims),
        "supported_claims": supported,
        "unsupported_claims": unsupported
    }

def extract_claims(response: str) -> list[str]:
    """Extract factual claims from response."""
    prompt = f"""Extract all factual claims from this response.
Each claim should be atomic and verifiable.

Response: {response}

Claims (one per line):"""

    result = llm.generate(prompt)
    return [c.strip() for c in result.split("\n") if c.strip()]

def is_supported_by_context(claim: str, contexts: list[str]) -> bool:
    """Check if claim is supported by any context."""
    combined_context = "\n".join(contexts)

    prompt = f"""Is this claim supported by the context?

Claim: {claim}

Context: {combined_context}

Answer SUPPORTED or NOT_SUPPORTED:"""

    return "SUPPORTED" in llm.generate(prompt)
```

def answer_relevance(query: str, response: str) -> float:
    """Does the response actually answer the query?"""
    prompt = f"""Rate how well this response answers the query.

Query: {query}

Response: {response}

Score (0-10, where 10 = perfectly answers the question):"""

    result = llm.generate(prompt)
    return float(result.strip()) / 10

def answer_completeness(query: str, response: str, expected_points: list[str]) -> float:
    """Are all expected points covered?"""
    covered = 0

    for point in expected_points:
        if point_covered_in_response(point, response):
            covered += 1

    return covered / len(expected_points) if expected_points else 1.0

## Running Evaluations

### Batch Evaluation

```python
class RAGEvaluationRunner:
    def __init__(self, rag_pipeline, evaluator):
        self.pipeline = rag_pipeline
        self.evaluator = evaluator

    def evaluate_batch(self, eval_set: list[EvalExample]) -> dict:
        results = []

        for example in tqdm(eval_set):
            # Run RAG pipeline
            rag_result = self.pipeline.query(example.query)

            # Evaluate
            eval_result = self.evaluator.evaluate(
                query=example.query,
                response=rag_result["response"],
                contexts=rag_result["contexts"],
                ground_truth=example.expected_answer
            )

            # Add retrieval metrics
            eval_result["recall@10"] = recall_at_k(
                rag_result["doc_ids"],
                example.expected_docs,
                10
            )

            results.append(eval_result)

        return self.aggregate_results(results)

    def aggregate_results(self, results: list[dict]) -> dict:
        """Compute aggregate metrics."""
        metrics = {}

        for key in results[0].keys():
            values = [r[key] for r in results if isinstance(r.get(key), (int, float))]
            if values:
                metrics[key] = {
                    "mean": np.mean(values),
                    "std": np.std(values),
                    "min": np.min(values),
                    "max": np.max(values),
                }

        return metrics
```
