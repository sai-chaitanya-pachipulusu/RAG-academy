## The Evaluation Challenge

RAG systems fail in subtle ways:

- Retrieval can miss relevant docs
- Context can be noisy
- Generation can hallucinate
- End-to-end quality is hard to measure

## Evaluation Framework

### Three-Level Evaluation

```
┌─────────────────────────────────────────┐
│      Level 3: End-to-End Quality        │
│   (Answer quality, user satisfaction)    │
├─────────────────────────────────────────┤
│       Level 2: Generation Quality        │
│   (Faithfulness, relevance, coherence)   │
├─────────────────────────────────────────┤
│       Level 1: Retrieval Quality         │
│     (Recall, precision, relevance)       │
└─────────────────────────────────────────┘
```

## Complexity Analysis

| Level | Metrics | Time Complexity | Typical Latency |
|-------|---------|-----------------|-----------------|
| L1: Retrieval | Recall, Precision, MRR, nDCG | O(Q × K) | < 1s |
| L2: Generation | Faithfulness, Relevance | O(Q × L³) | 5-30s |
| L3: End-to-End | User satisfaction, Task success | O(Q × T) | Hours |

Where Q = queries, K = top-k, L = sequence length, T = task time

```python
from typing import List, Dict, Optional, Callable
from dataclasses import dataclass
import numpy as np
from datetime import datetime

@dataclass
class RAGResult:
    """Result from a RAG pipeline execution."""
    query: str
    contexts: List[str]
    context_ids: List[str]
    answer: str
    latency_ms: float
    token_usage: Dict[str, int]

@dataclass
class EvalExample:
    """Evaluation example with ground truth."""
    query: str
    expected_answer: Optional[str] = None
    expected_context_ids: Optional[List[str]] = None
    metadata: Optional[Dict] = None


class RAGEvaluationSuite:
    """
    Comprehensive RAG evaluation suite.
    
    Evaluates at three levels:
    1. Retrieval metrics
    2. Generation metrics
    3. End-to-end metrics
    """
    
    def __init__(
        self,
        judge_fn: Optional[Callable[[str], str]] = None
    ):
        """
        Initialize evaluation suite.
        
        Args:
            judge_fn: LLM function for generation evaluation
        """
        self.judge_fn = judge_fn
        self.history: List[Dict] = []
    
    # ==================== Level 1: Retrieval Metrics ====================
    
    def recall_at_k(
        self,
        retrieved_ids: List[str],
        relevant_ids: set,
        k: int = 10
    ) -> float:
        """Recall@K: Fraction of relevant docs found."""
        if not relevant_ids:
            return 1.0
        top_k = set(retrieved_ids[:k])
        return len(top_k & relevant_ids) / len(relevant_ids)
    
    def precision_at_k(
        self,
        retrieved_ids: List[str],
        relevant_ids: set,
        k: int = 10
    ) -> float:
        """Precision@K: Fraction of top-K that are relevant."""
        if not retrieved_ids:
            return 0.0
        top_k = retrieved_ids[:k]
        return sum(1 for d in top_k if d in relevant_ids) / len(top_k)
    
    def mrr(self, retrieved_ids: List[str], relevant_ids: set) -> float:
        """MRR: Reciprocal rank of first relevant document."""
        for i, doc_id in enumerate(retrieved_ids, start=1):
            if doc_id in relevant_ids:
                return 1.0 / i
        return 0.0
    
    # ==================== Level 2: Generation Metrics ====================
    
    def faithfulness_score(
        self,
        answer: str,
        contexts: List[str]
    ) -> Dict:
        """
        Evaluate if answer is grounded in contexts.
        
        Uses LLM-as-a-Judge if available, otherwise heuristics.
        """
        if self.judge_fn:
            prompt = f"""Rate how well this answer is supported by the context (0-1):

Context: {' '.join(contexts)[:1000]}
Answer: {answer}

Respond with JSON: {{"score": X.XX, "reasoning": "..."}}"""
            
            try:
                response = self.judge_fn(prompt)
                import json
                result = json.loads(response)
                return {
                    "score": float(result.get("score", 0.5)),
                    "reasoning": result.get("reasoning", ""),
                    "method": "llm_judge"
                }
            except:
                pass
        
        # Fallback: keyword overlap
        answer_words = set(answer.lower().split())
        context_words = set(' '.join(contexts).lower().split())
        overlap = len(answer_words & context_words) / len(answer_words) if answer_words else 1.0
        
        return {
            "score": overlap,
            "reasoning": "Keyword overlap fallback",
            "method": "heuristic"
        }
    
    def relevance_score(self, query: str, answer: str) -> Dict:
        """Evaluate if answer addresses the query."""
        if self.judge_fn:
            prompt = f"""Rate how well this answer addresses the query (0-1):

Query: {query}
Answer: {answer}

Respond with JSON: {{"score": X.XX, "reasoning": "..."}}"""
            
            try:
                response = self.judge_fn(prompt)
                import json
                result = json.loads(response)
                return {
                    "score": float(result.get("score", 0.5)),
                    "reasoning": result.get("reasoning", ""),
                    "method": "llm_judge"
                }
            except:
                pass
        
        # Fallback: keyword overlap
        query_words = set(query.lower().split())
        answer_words = set(answer.lower().split())
        overlap = len(query_words & answer_words) / len(query_words) if query_words else 1.0
        
        return {
            "score": overlap,
            "reasoning": "Keyword overlap fallback",
            "method": "heuristic"
        }
    
    # ==================== Level 3: End-to-End Metrics ====================
    
    def answer_correctness(
        self,
        generated: str,
        expected: str
    ) -> Dict:
        """Compare generated answer to expected answer."""
        # Simple word overlap (in production, use LLM or semantic similarity)
        gen_words = set(generated.lower().split())
        exp_words = set(expected.lower().split())
        
        if not exp_words:
            return {"score": 1.0, "method": "exact"}
        
        precision = len(gen_words & exp_words) / len(gen_words) if gen_words else 0.0
        recall = len(gen_words & exp_words) / len(exp_words)
        
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0.0
        
        return {
            "score": f1,
            "precision": precision,
            "recall": recall,
            "method": "token_f1"
        }
    
    # ==================== Full Evaluation ====================
    
    def evaluate(
        self,
        result: RAGResult,
        example: EvalExample,
        k: int = 10
    ) -> Dict:
        """
        Run complete evaluation on a RAG result.
        
        Returns:
            Dict with all metrics
        """
        metrics = {
            "timestamp": datetime.utcnow().isoformat(),
            "query": result.query,
            "latency_ms": result.latency_ms,
            "token_usage": result.token_usage
        }
        
        # Level 1: Retrieval
        if example.expected_context_ids:
            relevant = set(example.expected_context_ids)
            metrics["retrieval"] = {
                f"recall@{k}": self.recall_at_k(result.context_ids, relevant, k),
                f"precision@{k}": self.precision_at_k(result.context_ids, relevant, k),
                "mrr": self.mrr(result.context_ids, relevant)
            }
        
        # Level 2: Generation
        metrics["generation"] = {
            "faithfulness": self.faithfulness_score(result.answer, result.contexts),
            "relevance": self.relevance_score(result.query, result.answer)
        }
        
        # Level 3: End-to-end
        if example.expected_answer:
            metrics["end_to_end"] = {
                "correctness": self.answer_correctness(
                    result.answer,
                    example.expected_answer
                )
            }
        
        # Store in history
        self.history.append(metrics)
        
        return metrics
    
    def aggregate(self, metrics_list: List[Dict]) -> Dict:
        """Aggregate metrics across multiple evaluations."""
        if not metrics_list:
            return {}
        
        # Extract numeric values
        retrieval_scores = []
        faithfulness_scores = []
        relevance_scores = []
        
        for m in metrics_list:
            if "retrieval" in m:
                retrieval_scores.append(m["retrieval"].get(f"recall@10", 0))
            if "generation" in m:
                faithfulness_scores.append(m["generation"]["faithfulness"]["score"])
                relevance_scores.append(m["generation"]["relevance"]["score"])
        
        return {
            "count": len(metrics_list),
            "retrieval": {
                "mean": np.mean(retrieval_scores) if retrieval_scores else 0,
                "std": np.std(retrieval_scores) if retrieval_scores else 0
            },
            "faithfulness": {
                "mean": np.mean(faithfulness_scores) if faithfulness_scores else 0,
                "std": np.std(faithfulness_scores) if faithfulness_scores else 0
            },
            "relevance": {
                "mean": np.mean(relevance_scores) if relevance_scores else 0,
                "std": np.std(relevance_scores) if relevance_scores else 0
            }
        }
```
