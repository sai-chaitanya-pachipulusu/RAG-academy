  Route queries to **fast / standard / slow** retrieval paths to control cost
  while preserving quality on hard queries.
</Callout>

## The "Overkill" Problem

Not all queries deserve the same compute:

- **Easy queries** (70%): Simple lookups, greetings, math
- **Medium queries** (20%): Standard RAG pipeline
- **Hard queries** (10%): Need expensive transforms (HyDE, multi-query, rerank)

Running the "slow path" on every query wastes resources and increases latency.

## Complexity Analysis

| Path | Latency | Cost/Query | Quality |
|------|---------|------------|---------|
| Fast | 50ms | $0.001 | Good for simple |
| Standard | 200ms | $0.005 | Balanced |
| Slow | 800ms | $0.02 | Best for complex |

```python
from typing import Dict, Callable, List
from dataclasses import dataclass
from enum import Enum
import re

class DifficultyLevel(Enum):
    """Query difficulty levels."""
    EASY = "easy"
    MEDIUM = "medium"
    HARD = "hard"

@dataclass
class RetrievalPath:
    """Configuration for a retrieval path."""
    name: str
    retriever: Callable[[str], List[str]]
    reranker: Callable[[str, List[str]], List[str]]
    max_chunks: int
    use_hyde: bool
    use_multi_query: bool


class DifficultyRouter:
    """
    Router that selects retrieval paths based on query difficulty.
    
    Routes easy queries to fast paths, hard queries to thorough paths.
    """
    
    def __init__(
        self,
        fast_path: RetrievalPath,
        standard_path: RetrievalPath,
        slow_path: RetrievalPath
    ):
        """
        Initialize with three retrieval paths.
        
        Args:
            fast_path: Quick retrieval for easy queries
            standard_path: Balanced retrieval for medium queries
            slow_path: Thorough retrieval for hard queries
        """
        self.fast_path = fast_path
        self.standard_path = standard_path
        self.slow_path = slow_path
        
        # Difficulty indicators
        self.easy_patterns = [
            r'^\s*\d+\s*[+\-*/]\s*\d+',  # Math
            r'^(hi|hello|hey|thanks|bye)\b',  # Greetings
            r'^(what is|who is|when is|where is)\s+\w+$',  # Simple facts
        ]
        
        self.hard_patterns = [
            r'\b(compare|contrast|analyze|evaluate)\b',
            r'\b(why|how does|explain the)\b',
            r'\b(trends?|patterns?|relationship)\b',
            r'\b(pros? and cons?|advantages? and disadvantages?)\b',
            r'\b(implications?|consequences?|impact)\b',
        ]
    
    def assess_difficulty(self, query: str) -> DifficultyLevel:
        """
        Assess query difficulty.
        
        Args:
            query: User query
        
        Returns:
            DifficultyLevel enum
        """
        query_lower = query.lower().strip()
        word_count = len(query_lower.split())
        
        # Check easy patterns
        for pattern in self.easy_patterns:
            if re.search(pattern, query_lower):
                return DifficultyLevel.EASY
        
        # Check hard patterns
        hard_indicators = 0
        for pattern in self.hard_patterns:
            if re.search(pattern, query_lower):
                hard_indicators += 1
        
        if hard_indicators >= 2 or word_count > 15:
            return DifficultyLevel.HARD
        elif hard_indicators == 1 or word_count > 8:
            return DifficultyLevel.MEDIUM
        
        return DifficultyLevel.EASY
    
    def select_path(self, query: str) -> RetrievalPath:
        """
        Select appropriate retrieval path.
        
        Args:
            query: User query
        
        Returns:
            Selected retrieval path
        """
        difficulty = self.assess_difficulty(query)
        
        if difficulty == DifficultyLevel.EASY:
            return self.fast_path
        elif difficulty == DifficultyLevel.HARD:
            return self.slow_path
        else:
            return self.standard_path
    
    def route(self, query: str) -> Dict:
        """
        Route query through selected path.
        
        Args:
            query: User query
        
        Returns:
            Dict with results and path info
        """
        path = self.select_path(query)
        difficulty = self.assess_difficulty(query)
        
        # Execute retrieval
        chunks = path.retriever(query)
        
        # Apply reranker if configured
        if path.reranker:
            chunks = path.reranker(query, chunks)
        
        # Limit chunks
        chunks = chunks[:path.max_chunks]
        
        return {
            'query': query,
            'difficulty': difficulty.value,
            'path': path.name,
            'chunks': chunks,
            'config': {
                'max_chunks': path.max_chunks,
                'use_hyde': path.use_hyde,
                'use_multi_query': path.use_multi_query
            }
        }
    
    def get_path_stats(self) -> Dict:
        """
        Get statistics about configured paths.
        
        Returns:
            Dict with path configurations
        """
        return {
            'fast': {
                'name': self.fast_path.name,
                'max_chunks': self.fast_path.max_chunks,
                'use_hyde': self.fast_path.use_hyde,
                'use_multi_query': self.fast_path.use_multi_query
            },
            'standard': {
                'name': self.standard_path.name,
                'max_chunks': self.standard_path.max_chunks,
                'use_hyde': self.standard_path.use_hyde,
                'use_multi_query': self.standard_path.use_multi_query
            },
            'slow': {
                'name': self.slow_path.name,
                'max_chunks': self.slow_path.max_chunks,
                'use_hyde': self.slow_path.use_hyde,
                'use_multi_query': self.slow_path.use_multi_query
            }
        }
```

        self,
        fast_path: RetrievalPath,
        standard_path: RetrievalPath,
        slow_path: RetrievalPath
    ):
        self.fast_path = fast_path
        self.standard_path = standard_path
        self.slow_path = slow_path
        
        self.easy_patterns = [
            r'^\s*\d+\s*[+\-*/]\s*\d+',
            r'^(hi|hello|hey|thanks|bye)\b',
            r'^(what is|who is|when is|where is)\s+\w+$',

            r'\b(compare|contrast|analyze|evaluate)\b',

            r'\b(trends?|patterns?|relationship)\b',

    def assess_difficulty(self, query: str) -> DifficultyLevel:

        query_lower = query.lower().strip()
        word_count = len(query_lower.split())
        
        for pattern in self.easy_patterns:
            if re.search(pattern, query_lower):
                return DifficultyLevel.EASY
        
        hard_indicators = sum(
            1 for p in self.hard_patterns if re.search(p, query_lower)
        )
        
        if hard_indicators >= 2 or word_count > 15:
            return DifficultyLevel.HARD
        elif hard_indicators == 1 or word_count > 8:
            return DifficultyLevel.MEDIUM
        
        return DifficultyLevel.EASY

        """Select appropriate retrieval path."""
        difficulty = self.assess_difficulty(query)
        
        if difficulty == DifficultyLevel.EASY:
            return self.fast_path
        elif difficulty == DifficultyLevel.HARD:
            return self.slow_path
        return self.standard_path
    
    def route(self, query: str) -> Dict:
        """Route query through selected path."""
        path = self.select_path(query)
        difficulty = self.assess_difficulty(query)
        
        chunks = path.retriever(query)

            chunks = path.reranker(query, chunks)

            'path': path.name,

            'config': {
                'max_chunks': path.max_chunks,
                'use_hyde': path.use_hyde,

            }

## Best Practices

1. **Monitor quality**: Ensure fast path maintains accuracy
2. **Fallback**: Route to slower path if fast path fails
3. **Learn from feedback**: Adjust difficulty thresholds
4. **A/B test**: Compare routing vs uniform approach
5. **Track metrics**: Latency, cost, accuracy per path
