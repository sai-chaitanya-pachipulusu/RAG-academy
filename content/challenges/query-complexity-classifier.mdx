## Why Query Classification Matters

Not all queries need the same treatment. Simple lookups, complex reasoning, and ambiguous questions each need different handling in a RAG system.

**Query classification enables:**

- **Adaptive routing**: Route simple queries to fast paths
- **Cost optimization**: Use cheaper models for easy queries
- **Quality preservation**: Use powerful models for complex queries
- **User experience**: Detect ambiguous queries and ask for clarification

## Complexity Levels

| Level | Description | Example | Handling |
|-------|-------------|---------|----------|
| **NO_RETRIEVAL** | LLM can answer directly | "What is 2 + 2?" | Skip retrieval |
| **SIMPLE** | Single retrieval pass | "What is our return policy?" | Standard RAG |
| **COMPLEX** | Multi-step reasoning | "Compare Q1 and Q2 performance" | Advanced RAG |
| **AMBIGUOUS** | Needs clarification | "it" | Ask user |

## Complexity Analysis

| Operation | Time | Space |
|-----------|------|-------|
| Pattern matching | O(n) | O(1) |
| Word count | O(n) | O(1) |
| Classification | O(n) | O(1) |

## Implementation

```python
from enum import Enum
from dataclasses import dataclass
from typing import List, Tuple
import re

class QueryComplexity(Enum):
    """Query complexity levels for RAG routing."""
    NO_RETRIEVAL = "no_retrieval"  # LLM can answer directly
    SIMPLE = "simple"               # Single retrieval pass
    COMPLEX = "complex"             # Multi-step reasoning needed
    AMBIGUOUS = "ambiguous"         # Needs clarification

@dataclass
class ClassificationResult:
    """Result of query classification."""
    complexity: QueryComplexity
    confidence: float
    reasoning: str


class QueryClassifier:
    """
    Rule-based query complexity classifier.
    
    Uses pattern matching to categorize queries by complexity.
    """
    
    def __init__(self):
        """Initialize classifier with rules and patterns."""
        self.no_retrieval_patterns: List[str] = []
        self.complex_patterns: List[str] = []
        self.ambiguous_patterns: List[str] = []
        
        self._build_patterns()
    
    def _build_patterns(self):
        """Build regex patterns for classification."""
        # NO_RETRIEVAL patterns
        self.no_retrieval_patterns = [
            r"^\d+\s*[+\-*/]\s*\d+",       # Math operations
            r"^(hi|hello|hey|greetings)",  # Greetings
            r"what (is|does) .+ stand for", # Acronyms
            r"^(what is|define|explain) (a|an|the)? ?\w+$",  # Simple definitions
            r"^(how are you|what's up|good morning|good afternoon)",  # Casual
        ]
        
        # COMPLEX patterns
        self.complex_patterns = [
            r"compare|comparison|versus|vs\.?",
            r"analyze|analysis|explain why",
            r"(pros? and cons?|advantages? and disadvantages?)",
            r"(trends?|patterns?|over time)",
            r"(factors?|reasons?|causes?) .*(led to|behind|for)",
            r"summarize .* and .*(identify|find)",
            r"relationship between",
            r"trade-offs?",
        ]
        
        # AMBIGUOUS patterns
        self.ambiguous_patterns = [
            r"^(it|this|that|those|these)$",
            r"^.{1,3}$",  # Very short (1-3 chars)
            r"^(the |that |this )?(thing|stuff|document|one)s?$",
            r"\b(what about|tell me about)\s*$",  # Incomplete
        ]
    
    def classify(self, query: str) -> ClassificationResult:
        """
        Classify query complexity.
        
        Args:
            query: User's question
            
        Returns:
            ClassificationResult with complexity, confidence, and reasoning
        """
        query = query.strip()
        query_lower = query.lower()
        
        # Check ambiguous first (short queries)
        is_amb, amb_conf, amb_reason = self._check_ambiguous(query)
        if is_amb and amb_conf > 0.7:
            return ClassificationResult(
                QueryComplexity.AMBIGUOUS, amb_conf, amb_reason
            )
        
        # Check no_retrieval
        is_no_ret, no_ret_conf, no_ret_reason = self._check_no_retrieval(query)
        if is_no_ret and no_ret_conf > 0.6:
            return ClassificationResult(
                QueryComplexity.NO_RETRIEVAL, no_ret_conf, no_ret_reason
            )
        
        # Check complex
        is_complex, complex_conf, complex_reason = self._check_complex(query)
        if is_complex and complex_conf > 0.5:
            return ClassificationResult(
                QueryComplexity.COMPLEX, complex_conf, complex_reason
            )
        
        # Default to simple
        return ClassificationResult(
            QueryComplexity.SIMPLE, 0.7, "Default classification for clear questions"
        )
    
    def _check_no_retrieval(self, query: str) -> Tuple[bool, float, str]:
        """
        Check if query can be answered without retrieval.
        
        Returns:
            (is_no_retrieval, confidence, reasoning)
        """
        query_lower = query.lower().strip()
        
        for pattern in self.no_retrieval_patterns:
            if re.search(pattern, query_lower):
                return True, 0.9, f"Matches pattern: {pattern}"
        
        # Check for very short factual questions
        words = query_lower.split()
        if len(words) <= 3 and any(w in query_lower for w in ['what', 'who', 'when', 'where']):
            return True, 0.7, "Short factual question"
        
        return False, 0.0, ""
    
    def _check_complex(self, query: str) -> Tuple[bool, float, str]:
        """
        Check if query requires complex multi-step reasoning.
        
        Returns:
            (is_complex, confidence, reasoning)
        """
        query_lower = query.lower().strip()
        
        matches = []
        for pattern in self.complex_patterns:
            if re.search(pattern, query_lower):
                matches.append(pattern)
        
        if len(matches) >= 2:
            return True, 0.9, f"Multiple complex indicators: {', '.join(matches[:2])}"
        elif len(matches) == 1:
            return True, 0.7, f"Complex indicator: {matches[0]}"
        
        # Check for long queries
        word_count = len(query_lower.split())
        if word_count > 15:
            return True, 0.6, f"Long query ({word_count} words)"
        
        return False, 0.0, ""
    
    def _check_ambiguous(self, query: str) -> Tuple[bool, float, str]:
        """
        Check if query is ambiguous and needs clarification.
        
        Returns:
            (is_ambiguous, confidence, reasoning)
        """
        query_lower = query.lower().strip()
        
        for pattern in self.ambiguous_patterns:
            if re.search(pattern, query_lower):
                return True, 0.95, f"Ambiguous pattern: {pattern}"
        
        # Check for pronouns without context
        pronouns = ['it', 'this', 'that', 'they', 'them']
        words = query_lower.split()
        if len(words) <= 2 and any(w in pronouns for w in words):
            return True, 0.8, "Pronoun without context"
        
        return False, 0.0, ""
```

def classify_with_llm_fallback(self, query: str) -> ClassificationResult:
    """
    Use rules first, fall back to LLM for uncertain cases.
    """
    result = self.classify(query)
    
    # If confidence is low, use LLM
    if result.confidence < 0.6:
        llm_result = self._llm_classify(query)

## Best Practices

1. **Start with rules**: Rules are fast and interpretable
2. **Add LLM fallback**: For edge cases and improved accuracy
3. **Monitor performance**: Track classification accuracy
4. **Iterate on patterns**: Add new patterns based on errors
5. **Consider context**: Use conversation history for ambiguous queries
