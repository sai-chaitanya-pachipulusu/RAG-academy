## The Conversational Challenge

Single-turn RAG:

```
User: "What is the refund policy?"
→ Retrieve → Generate → Done
```

Multi-turn RAG:

```
User: "What is the refund policy?"
→ Retrieve → Generate
User: "How long do I have?"  ← Need to remember we're talking about refunds!
→ Retrieve (with context) → Generate
```

## Why Conversational RAG Matters

| Aspect | Single-Turn | Multi-Turn |
|--------|-------------|------------|
| Context | None | Full conversation history |
| Pronouns | Fail | Resolved |
| Follow-ups | Treated independently | Connected |
| User Experience | Robotic | Natural |

## Conversation State Management

```python
from dataclasses import dataclass, field
from typing import List, Optional, Dict
import time

@dataclass
class Message:
    role: str  # "user" or "assistant"
    content: str
    timestamp: float = field(default_factory=time.time)
    retrieved_docs: List[str] = field(default_factory=list)

@dataclass
class Conversation:
    id: str
    messages: List[Message] = field(default_factory=list)
    metadata: Dict = field(default_factory=dict)
    
    def add_user_message(self, content: str):
        self.messages.append(Message(role="user", content=content))
    
    def add_assistant_message(self, content: str, docs: List[str] = None):
        self.messages.append(Message(
            role="assistant", 
            content=content,
            retrieved_docs=docs or []
        ))
    
    def get_history(self, n_turns: int = 5) -> str:
        """Get recent conversation history as string."""
        recent = self.messages[-n_turns*2:]  # n turns = 2n messages
        return "\n".join([f"{m.role}: {m.content}" for m in recent])
```

## Retrieval Strategies

### Strategy 1: Query Rewriting

```python
class ConversationalRetriever:
    def __init__(self, base_retriever, llm):
        self.retriever = base_retriever
        self.llm = llm
        self.conversations: Dict[str, Conversation] = {}
    
    def retrieve(self, query: str, conversation_id: str) -> List[dict]:
        """Retrieve with conversation context."""
        conv = self.conversations.get(conversation_id, Conversation(id=conversation_id))
        
        # Rewrite query with context
        contextualized = self.rewrite_query(query, conv)
        
        # Retrieve using contextualized query
        docs = self.retriever.search(contextualized)
        
        # Also retrieve for original query (for disambiguation)
        if contextualized != query:
            original_docs = self.retriever.search(query)
            docs = self.merge_results(docs, original_docs)
        
        return docs
    
    def rewrite_query(self, query: str, conversation: Conversation) -> str:
        """Rewrite query using conversation history."""
        if len(conversation.messages) < 2:
            return query
        
        history = conversation.get_history(n_turns=3)
        
        prompt = f"""Rewrite this follow-up question to be self-contained.

Conversation history:
{history}

Follow-up: "{query}"

Rewrite (make references explicit):"""
        
        rewritten = self.llm.generate(prompt, max_tokens=100)
        return rewritten.strip() or query
```

### Strategy 2: Contextualized Embeddings

```python
class ContextualizedEmbedder:
    """Create embeddings that include conversation context."""
    
    def __init__(self, base_embedder):
        self.embedder = base_embedder
    
    def embed_query(self, query: str, conversation: Conversation) -> List[float]:
        """Embed query with conversation context."""
        # Get conversation summary
        summary = self.summarize_conversation(conversation)
        
        # Combine with query
        contextualized = f"Context: {summary}\nQuery: {query}"
        
        return self.embedder.encode(contextualized)
    
    def summarize_conversation(self, conversation: Conversation) -> str:
        """Extract key topics from conversation."""
        if len(conversation.messages) < 2:
            return ""
        
        # Simple extraction: use last user message
        user_messages = [m for m in conversation.messages if m.role == "user"]
        if user_messages:
            return user_messages[-1].content[:100]
        return ""
```

### Strategy 3: Multi-Query Fusion

```python
class MultiQueryConversationalRetriever:
    """Generate multiple queries from conversation context."""
    
    def retrieve(self, query: str, conversation: Conversation) -> List[dict]:
        """Retrieve using multiple generated queries."""
        # Generate variations
        variations = self.generate_query_variations(query, conversation)
        
        # Retrieve for each
        all_results = []
        for var in variations:
            results = self.retriever.search(var)
            all_results.extend(results)
        
        # Deduplicate and rerank
        return self.deduplicate_and_rerank(all_results, query)
    
    def generate_query_variations(
        self, 
        query: str, 
        conversation: Conversation
    ) -> List[str]:
        """Generate query variations based on context."""
        variations = [query]  # Always include original
        
        # Add rewritten version
        rewritten = self.rewrite_query(query, conversation)
        if rewritten != query:
            variations.append(rewritten)
        
        # Add entity-focused variations
        entities = self.extract_entities(conversation)
        for entity in entities[:2]:
            variations.append(f"{query} {entity}")
        
        return variations[:4]  # Limit to avoid too many calls
```

## Conversation Memory Integration

```python
class ConversationalRAG:
    """Full conversational RAG system."""
    
    def __init__(self, retriever, llm, memory_store):
        self.retriever = retriever
        self.llm = llm
        self.memory = memory_store
    
    def chat(self, query: str, conversation_id: str) -> dict:
        """Process a conversational query."""
        # Load or create conversation
        conversation = self.memory.load(conversation_id)
        
        # Add user message
        conversation.add_user_message(query)
        
        # Retrieve with context
        docs = self.retrieve_with_context(query, conversation)
        
        # Generate response
        response = self.generate_response(query, docs, conversation)
        
        # Add assistant message
        conversation.add_assistant_message(
            response, 
            docs=[d["id"] for d in docs]
        )
        
        # Save conversation
        self.memory.save(conversation)
        
        return {
            "response": response,
            "conversation_id": conversation_id,
            "documents_used": [d["id"] for d in docs]
        }
    
    def retrieve_with_context(
        self, 
        query: str, 
        conversation: Conversation
    ) -> List[dict]:
        """Retrieve documents using conversation context."""
        # Rewrite query
        rewritten = self.rewrite_query(query, conversation)
        
        # Retrieve
        docs = self.retriever.search(rewritten)
        
        # Filter by conversation context
        docs = self.filter_by_context(docs, conversation)
        
        return docs
    
    def generate_response(
        self, 
        query: str, 
        docs: List[dict], 
        conversation: Conversation
    ) -> str:
        """Generate response with conversation context."""
        # Build full context
        history = conversation.get_history(n_turns=3)
        doc_context = self.format_documents(docs)
        
        prompt = f"""You are a helpful assistant. Use the conversation history and retrieved documents to answer.

Conversation History:
{history}

Retrieved Information:
{doc_context}

User: {query}
Assistant:"""
        
        return self.llm.generate(prompt)
```

def needs_clarification(query: str, conversation: Conversation) -> bool:
    """Check if query needs clarification."""
    # Ambiguous pronouns without clear referent
    ambiguous_pronouns = ["it", "they", "them", "this", "that"]
    
    has_ambiguous_pronoun = any(
        pronoun in query.lower() for pronoun in ambiguous_pronouns
    )
    
    if not has_ambiguous_pronoun:
        return False
    
    # Check if we can resolve from context
    referent = resolve_pronoun(query, conversation)
    
    return referent is None

## Best Practices

1. **Keep recent context**: Last 3-5 turns are most important
2. **Summarize older turns**: Compress distant history
3. **Track entities**: Resolve pronouns and references
4. **Detect topic switches**: Reset context when needed
5. **Handle clarifications**: Ask when uncertain
