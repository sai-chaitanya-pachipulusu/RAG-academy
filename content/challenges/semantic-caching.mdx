# Semantic Caching

Implement semantic caching to avoid redundant LLM calls by matching semantically similar queries, achieving significant cost and latency savings.

## Why Semantic Caching?

LLM calls are:
- **Expensive**: $0.01-0.10 per query
- **Slow**: 500ms-3s per call
- **Redundant**: Many queries are semantically similar

Semantic caching detects when a new query is similar to a cached one and returns the cached response instantly.

## Exact Match vs Semantic Match

```python
# Exact match (basic)
cache = {"What is the capital of France?": "Paris"}
query = "What's the capital of France?"  # MISS! Different string

# Semantic match
query_embedding = embed("What's the capital of France?")
similar = find_similar(query_embedding, threshold=0.95)
# HIT! "What is the capital of France?" is 0.97 similar
```

## Algorithm Details

### Semantic Cache Implementation

```python
class SemanticCache:
    def __init__(self, embedder, similarity_threshold=0.95, 
                 max_size=10000, ttl_seconds=None):
        self.embedder = embedder
        self.threshold = similarity_threshold
        self.max_size = max_size
        self.ttl = ttl_seconds
        
        self.cache: Dict[str, CacheEntry] = {}
        self.embeddings: List[np.ndarray] = []
        self.keys: List[str] = []
        self.stats = {"hits": 0, "misses": 0}
    
    def get(self, query: str) -> Optional[str]:
        """Check cache for semantically similar query."""
        # Clean expired entries periodically
        if self.ttl and len(self.cache) % 100 == 0:
            self.invalidate_expired()
        
        # Embed query
        query_embedding = self.embedder.encode(query)
        
        # Find similar
        similar = self.find_similar(query_embedding)
        
        if similar:
            self.stats["hits"] += 1
            similar.access_count += 1
            return similar.response
        
        self.stats["misses"] += 1
        return None
    
    def find_similar(self, query_embedding: np.ndarray) -> Optional[CacheEntry]:
        """Find most similar cached query."""
        if not self.embeddings:
            return None
        
        # Compute similarities
        cached_matrix = np.array(self.embeddings)
        similarities = np.dot(query_embedding, cached_matrix.T)
        
        # Find best match
        max_idx = np.argmax(similarities)
        max_sim = similarities[max_idx]
        
        if max_sim >= self.threshold:
            cached_key = self.keys[max_idx]
            return self.cache[cached_key]
        
        return None
    
    def set(self, query: str, response: str) -> None:
        """Add to cache."""
        import time
        
        # Embed query
        embedding = self.embedder.encode(query)
        
        # Create entry
        entry = CacheEntry(
            query=query,
            query_embedding=embedding,
            response=response,
            timestamp=time.time()
        )
        
        # Store
        key = hash(query)  # Simple key
        self.cache[key] = entry
        self.embeddings.append(embedding)
        self.keys.append(key)
        
        # Evict if needed
        if len(self.cache) > self.max_size:
            self._evict_oldest()
    
    def _evict_oldest(self):
        """Remove oldest entry."""
        import time
        
        oldest_key = min(
            self.cache.keys(),
            key=lambda k: self.cache[k].timestamp
        )
        
        idx = self.keys.index(oldest_key)
        del self.cache[oldest_key]
        del self.embeddings[idx]
        del self.keys[idx]
```

### Cached LLM

```python
class CachedLLM:
    def __init__(self, llm_client, cache: SemanticCache):
        self.llm = llm_client
        self.cache = cache
    
    def generate(self, query: str, **kwargs) -> dict:
        """Generate with caching."""
        # Check cache
        cached_response = self.cache.get(query)
        
        if cached_response:
            # Find similarity for reporting
            query_emb = self.cache.embedder.encode(query)
            similar = self.cache.find_similar(query_emb)
            
            return {
                "response": cached_response,
                "cache_hit": True,
                "similarity": float(np.dot(
                    query_emb, 
                    similar.query_embedding
                )) if similar else None
            }
        
        # Cache miss - call LLM
        response = self.llm.generate(query, **kwargs)
        
        # Store in cache
        self.cache.set(query, response)
        
        return {
            "response": response,
            "cache_hit": False,
            "similarity": None
        }
```

## Threshold Selection

| Threshold | Behavior | Best For |
|-----------|----------|----------|
| 0.90 | More hits, some false positives | Tolerant applications |
| 0.95 | Balanced | General use |
| 0.98 | Fewer hits, high precision | Critical applications |

## Production Implementation with Vector DB

```python
from qdrant_client import QdrantClient
from qdrant_client.models import Distance, VectorParams, PointStruct

class VectorDBSemanticCache:
    """Production semantic cache using Qdrant."""
    
    def __init__(self, embedder, threshold=0.95, ttl_seconds=3600):
        self.embedder = embedder
        self.threshold = threshold
        self.ttl = ttl_seconds
        
        self.client = QdrantClient(":memory:")  # Or host:port
        self.client.create_collection(
            collection_name="semantic_cache",
            vectors_config=VectorParams(
                size=384,  # Embedding dimension
                distance=Distance.COSINE
            )
        )
    
    def get(self, query: str) -> Optional[str]:
        query_embedding = self.embedder.encode(query)
        
        # Search similar
        results = self.client.search(
            collection_name="semantic_cache",
            query_vector=query_embedding.tolist(),
            limit=1,
            score_threshold=self.threshold
        )
        
        if results:
            return results[0].payload["response"]
        return None
    
    def set(self, query: str, response: str):
        import uuid
        
        embedding = self.embedder.encode(query)
        
        self.client.upsert(
            collection_name="semantic_cache",
            points=[PointStruct(
                id=str(uuid.uuid4()),
                vector=embedding.tolist(),
                payload={
                    "query": query,
                    "response": response,
                    "timestamp": time.time()
                }
            )]
        )
```

## Evaluation Criteria

1. **Similarity Matching**: Correctly identifies semantically similar queries
2. **Threshold Respect**: Only returns cache hits above threshold
3. **Eviction**: Proper LRU/LFU eviction when full
4. **TTL Support**: Expires old entries correctly

## Test Cases

```python
def test_semantic_match():
    """Test semantically similar queries match."""
    cache = SemanticCache(embedder, threshold=0.90)
    
    cache.set("What is the capital of France?", "Paris")
    result = cache.get("What's France's capital?")
    
    assert result == "Paris"
    assert cache.get_stats()["hits"] == 1

def test_threshold_respected():
    """Test threshold prevents false positives."""
    cache = SemanticCache(embedder, threshold=0.98)
    
    cache.set("What is Python?", "A programming language")
    result = cache.get("What is JavaScript?")
    
    assert result is None  # Different concepts

def test_exact_match():
    """Test exact query strings match."""
    cache = SemanticCache(embedder)
    
    cache.set("test query", "test response")
    result = cache.get("test query")
    
    assert result == "test response"
```
