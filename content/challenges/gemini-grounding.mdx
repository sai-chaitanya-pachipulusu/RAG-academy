# Gemini Grounding

Gemini 2.0 can automatically ground responses in Google Search results, providing real-time information without building your own RAG pipeline.

**Your task**: Build a grounded Q&A system using Gemini's native grounding features.

def search(self, query: str) -> List[GroundingChunk]:
    if self.mock_mode:
        # Find best matching mock result
        query_terms = set(query.lower().split())
        best_match = None
        best_score = 0
        
        for mock_query, results in MOCK_SEARCH_RESULTS.items():
            mock_terms = set(mock_query.lower().split())
            score = len(query_terms & mock_terms)
            if score > best_score:
                best_score = score
                best_match = results
        
        return best_match or []
    else:
        # Real Gemini API call
        return self._real_gemini_search(query)

<summary>Hint 3: Grounded synthesis</summary>

Build response strictly from sources:

```python
def synthesize_grounded_response(self, user_query: str, chunks: List[GroundingChunk]) -> str:
    if not chunks:
        return "I couldn't find relevant information to answer this question."
    
    # Extract key facts from snippets
    facts = []
    for chunk in chunks:
        facts.append(f"{chunk.snippet} [{chunk.title}]")
    
    # Combine facts into coherent response
    response_parts = []
    for fact in facts:
        response_parts.append(fact)
    
    return " ".join(response_parts)
```

## Bonus: Real Gemini API

If you have a Google AI API key:

```python
import google.generativeai as genai

def real_gemini_grounded_search(query: str) -> GroundedResponse:
    genai.configure(api_key="YOUR_API_KEY")
    model = genai.GenerativeModel("gemini-2.0-flash")
    
    response = model.generate_content(
        query,
        tools=[{
            "google_search_retrieval": {
                "dynamic_retrieval_config": {
                    "mode": "MODE_DYNAMIC",
                    "dynamic_threshold": 0.3
                }
            }
        }]
    )
    
    # Extract grounding metadata
    chunks = []
    if response.candidates[0].grounding_metadata:
        for chunk in response.candidates[0].grounding_metadata.grounding_chunks:
            chunks.append(GroundingChunk(
                uri=chunk.web.uri,
                title=chunk.web.title,
                snippet=chunk.web.snippet
            ))
    
    return GroundedResponse(
        text=response.text,
        grounding_chunks=chunks,
        search_queries_used=[query]
    )
```
