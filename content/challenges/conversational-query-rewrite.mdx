# Conversational Query Rewriting

## The Problem with Follow-ups

**Conversation**:

```
User: "Tell me about Pinecone"
AI: "Pinecone is a vector database..."
User: "What about its pricing?"  ← Has "its" pronoun!
```

If we search for "What about its pricing?", retrieval fails. We need "What is Pinecone's pricing?"

---

## The Rewriting Solution

Before retrieval, rewrite the query to be **standalone**:

```python
history = [
    {"role": "user", "content": "Tell me about Pinecone"},
    {"role": "assistant", "content": "Pinecone is a vector database..."}
]
current = "What about its pricing?"

rewritten = "What is Pinecone's pricing?"
```

Now retrieval works!

---

## Pronoun Resolution

| Pronoun         | Resolution                    | Example Context |
| --------------- | ----------------------------- | --------------- |
| "it", "its"     | Most recent noun from history | "Pinecone... its pricing" |
| "they", "their" | Most recent plural entity     | "OpenAI... their API" |
| "this", "that"  | Most recent topic             | "RAG... this approach" |
| "he", "she"     | Most recent person            | "Elon Musk... he said" |

## Implementation Approaches

### Rule-Based Rewriting

```python
import re
from typing import List, Dict, Optional

class RuleBasedRewriter:
    """Fast rule-based query rewriting."""

    def __init__(self):
        self.pronoun_patterns = {
            r'\bit\b': 'SUBJECT',
            r'\bits\b': 'POSSESSIVE',
            r'\bthis\b': 'TOPIC',
            r'\bthat\b': 'TOPIC',
            r'\bthey\b': 'PLURAL',
            r'\btheir\b': 'PLURAL_POSSESSIVE',
        }

    def rewrite(
        self,
        query: str,
        conversation_history: List[Dict]
    ) -> str:
        """
        Rewrite query using conversation context.

        Time Complexity: O(n) where n = history length
        Space Complexity: O(1)
        """
        # Extract entities from history
        entities = self.extract_entities(conversation_history)

        if not entities:
            return query

        # Find pronouns in query
        rewritten = query

        for pattern, entity_type in self.pronoun_patterns.items():
            if re.search(pattern, query, re.IGNORECASE):
                # Find matching entity
                entity = self.find_matching_entity(entities, entity_type)
                if entity:
                    rewritten = re.sub(
                        pattern,
                        entity,
                        rewritten,
                        flags=re.IGNORECASE
                    )

        return rewritten

    def extract_entities(self, history: List[Dict]) -> List[Dict]:
        """Extract entities from conversation history."""
        entities = []

        for msg in history:
            content = msg.get("content", "")

            # Simple noun phrase extraction
            # In production, use NER (spaCy, etc.)
            words = content.split()
            for i, word in enumerate(words):
                if word[0].isupper() and len(word) > 2:
                    # Potential proper noun
                    entities.append({
                        "text": word,
                        "type": "PROPER_NOUN",
                        "position": i
                    })

        return entities

    def find_matching_entity(
        self,
        entities: List[Dict],
        entity_type: str
    ) -> Optional[str]:
        """Find entity matching the pronoun type."""
        if not entities:
            return None

        # Return most recent entity (last in list)
        return entities[-1]["text"]
```

### LLM-Based Rewriting

```python
class LLMRewriter:
    """LLM-based conversational query rewriting."""

    REWRITE_PROMPT = """Given the conversation history and a follow-up question, rewrite the question to be self-contained.

Conversation History:
{history}

Follow-up Question: {query}

Rewrite the follow-up question to make all references explicit and clear. Replace pronouns like "it", "they", "this" with the actual entities they refer to.

Rewritten Question:"""

    def __init__(self, llm_client):
        self.llm = llm_client

    def rewrite(
        self,
        query: str,
        conversation_history: List[Dict]
    ) -> str:
        """
        Rewrite query using LLM.

        Time Complexity: O(L) where L = LLM latency
        Space Complexity: O(1)
        """
        # Format history
        history_str = self._format_history(conversation_history)

        # Create prompt
        prompt = self.REWRITE_PROMPT.format(
            history=history_str,
            query=query
        )

        # Generate rewrite
        rewritten = self.llm.generate(prompt, max_tokens=150)

        # Clean up
        rewritten = rewritten.strip().strip('"')

        # Fallback to original if empty
        return rewritten if rewritten else query

    def _format_history(self, history: List[Dict]) -> str:
        """Format conversation history for prompt."""
        lines = []
        for msg in history[-6:]:  # Last 3 exchanges
            role = msg.get("role", "user").capitalize()
            content = msg.get("content", "")
            lines.append(f"{role}: {content}")
        return "\n".join(lines)
```

### Hybrid Rewriting (Production)

```python
class HybridRewriter:
    """
    Hybrid rewriter combining speed and accuracy.

    Uses rules for simple cases, LLM for complex ones.
    """

    def __init__(self, llm_client, confidence_threshold: float = 0.8):
        self.rule_rewriter = RuleBasedRewriter()
        self.llm_rewriter = LLMRewriter(llm_client)
        self.threshold = confidence_threshold

    def rewrite(
        self,
        query: str,
        conversation_history: List[Dict]
    ) -> Dict:
        """
        Rewrite with approach selection.

        Returns dict with rewritten query and metadata.
        """
        # Check if query has pronouns
        has_pronouns = self._has_pronouns(query)

        if not has_pronouns:
            return {
                "rewritten": query,
                "method": "none",
                "confidence": 1.0
            }

        # Try rule-based first (fast)
        rule_result = self.rule_rewriter.rewrite(query, conversation_history)
        rule_confidence = self._estimate_confidence(rule_result, query)

        if rule_confidence >= self.threshold:
            return {
                "rewritten": rule_result,
                "method": "rule",
                "confidence": rule_confidence
            }

        # Fall back to LLM
        llm_result = self.llm_rewriter.rewrite(query, conversation_history)

        return {
            "rewritten": llm_result,
            "method": "llm",
            "confidence": 0.9  # LLM generally high confidence
        }

    def _has_pronouns(self, query: str) -> bool:
        """Check if query contains pronouns needing resolution."""
        pronouns = ['it', 'its', 'they', 'their', 'this', 'that', 'he', 'she', 'him', 'her']
        query_lower = query.lower()
        return any(f' {p} ' in f' {query_lower} ' for p in pronouns)

    def _estimate_confidence(self, result: str, original: str) -> float:
        """Estimate confidence of rule-based rewrite."""
        # If unchanged, low confidence
        if result == original:
            return 0.3

        # If significantly changed, higher confidence
        change_ratio = len(result) / max(len(original), 1)
        if 0.8 <= change_ratio <= 2.0:
            return 0.85

        return 0.6
```

### Coreference Resolution

```python
class CoreferenceResolver:
    """Advanced coreference resolution for complex references."""

    def __init__(self, nlp_model):
        self.nlp = nlp_model  # spaCy with coreference

    def resolve(self, query: str, history: List[Dict]) -> str:
        """Resolve all coreferences in query."""
        # Combine history and query
        full_text = self._combine_text(history, query)

        # Process with NLP
        doc = self.nlp(full_text)

        # Get coreference clusters
        if hasattr(doc, '_'):
            clusters = doc._.coref_clusters

            # Build replacement map
            replacements = {}
            for cluster in clusters:
                # Main mention (first or most representative)
                main_mention = cluster.main.text

                for mention in cluster.mentions:
                    if mention.text.lower() in ['it', 'they', 'this', 'that']:
                        replacements[mention.text] = main_mention

            # Apply replacements
            resolved = query
            for old, new in replacements.items():
                resolved = resolved.replace(old, new)

            return resolved

        return query
```

## Tradeoffs

### Rule-Based

- ✅ Fast (no LLM call)
- ✅ Deterministic
- ✅ No API costs
- ❌ Limited patterns
- ❌ No semantic understanding

### LLM-Based

- ✅ Handles complex cases
- ✅ Semantic understanding
- ✅ Natural rewrites
- ❌ Adds latency (+200ms)
- ❌ Cost per query
- ❌ May hallucinate

### Hybrid

- ✅ Best of both worlds
- ✅ Scalable
- ✅ Cost-effective
- ❌ More complex
- ❌ Needs tuning
