## Why A/B Test RAG?

RAG has many tunable parameters:

- Chunk size (300 vs 500 vs 1000 tokens)
- Retrieval k (5 vs 10 vs 20)
- Reranking (none vs cross-encoder vs LLM)
- Generation model (GPT-4o-mini vs GPT-4o)
- Prompt templates

Intuition often fails. Data-driven decisions win.

## Complexity Analysis

| Operation | Time Complexity | Space Complexity | Sample Size |
|-----------|-----------------|------------------|-------------|
| Variant assignment | O(1) | O(1) | - |
| Metric collection | O(Q) | O(Q) | Q queries |
| Statistical test | O(Q) | O(Q) | Q queries |
| Full experiment | O(Q × V) | O(Q × V) | Q × V samples |

Where Q = queries, V = variants

```python
from typing import Dict, List, Callable, Optional
from dataclasses import dataclass
import hashlib
import random
from datetime import datetime
import statistics
from scipy import stats

@dataclass
class Variant:
    """Experiment variant configuration."""
    name: str
    weight: float  # 0.0 to 1.0
    config: Dict   # RAG configuration

@dataclass
class Experiment:
    """A/B test experiment definition."""
    name: str
    variants: List[Variant]
    metrics: List[str]
    start_date: datetime
    end_date: Optional[datetime] = None


class ABTestFramework:
    """
    Framework for A/B testing RAG configurations.
    
    Features:
    - Consistent user assignment
    - Metric collection
    - Statistical analysis
    - Winner selection
    """
    
    def __init__(self):
        self.experiments: Dict[str, Experiment] = {}
        self.assignments: Dict[str, Dict] = {}  # user_id -> {exp: variant}
        self.metrics: Dict[str, List[Dict]] = {}  # exp_name -> metrics
    
    def create_experiment(
        self,
        name: str,
        variants: List[Variant],
        metrics: List[str]
    ) -> Experiment:
        """Create a new experiment."""
        # Normalize weights
        total_weight = sum(v.weight for v in variants)
        for v in variants:
            v.weight = v.weight / total_weight
        
        exp = Experiment(
            name=name,
            variants=variants,
            metrics=metrics,
            start_date=datetime.utcnow()
        )
        self.experiments[name] = exp
        self.metrics[name] = []
        return exp
    
    def assign_variant(
        self,
        user_id: str,
        experiment_name: str
    ) -> Variant:
        """
        Deterministically assign user to variant.
        
        Uses hash for consistent assignment across sessions.
        """
        exp = self.experiments.get(experiment_name)
        if not exp:
            raise ValueError(f"Experiment {experiment_name} not found")
        
        # Check if already assigned
        if user_id in self.assignments:
            assigned = self.assignments[user_id].get(experiment_name)
            if assigned:
                for v in exp.variants:
                    if v.name == assigned:
                        return v
        
        # Deterministic assignment using hash
        hash_input = f"{experiment_name}:{user_id}"
        hash_value = int(hashlib.md5(hash_input.encode()).hexdigest(), 16)
        bucket = (hash_value % 10000) / 10000  # 0.0 to 0.9999
        
        cumulative = 0.0
        for variant in exp.variants:
            cumulative += variant.weight
            if bucket < cumulative:
                # Store assignment
                if user_id not in self.assignments:
                    self.assignments[user_id] = {}
                self.assignments[user_id][experiment_name] = variant.name
                return variant
        
        # Fallback to last variant
        return exp.variants[-1]
    
    def log_metric(
        self,
        experiment_name: str,
        user_id: str,
        variant_name: str,
        metric_name: str,
        value: float
    ):
        """Log a metric for analysis."""
        if experiment_name not in self.metrics:
            self.metrics[experiment_name] = []
        
        self.metrics[experiment_name].append({
            "user_id": user_id,
            "variant": variant_name,
            "metric": metric_name,
            "value": value,
            "timestamp": datetime.utcnow()
        })
    
    def analyze_experiment(
        self,
        experiment_name: str,
        metric_name: str,
        control_variant: str = "control"
    ) -> Dict:
        """
        Analyze experiment results.
        
        Returns:
            Dict with statistical analysis
        """
        data = self.metrics.get(experiment_name, [])
        
        # Group by variant
        by_variant: Dict[str, List[float]] = {}
        for record in data:
            if record["metric"] == metric_name:
                variant = record["variant"]
                if variant not in by_variant:
                    by_variant[variant] = []
                by_variant[variant].append(record["value"])
        
        if control_variant not in by_variant:
            return {"error": f"Control variant {control_variant} not found"}
        
        control_data = by_variant[control_variant]
        results = {
            "experiment": experiment_name,
            "metric": metric_name,
            "control": self._compute_stats(control_data),
            "variants": {}
        }
        
        # Compare each variant to control
        for variant_name, variant_data in by_variant.items():
            if variant_name == control_variant:
                continue
            
            comparison = self._compare_groups(control_data, variant_data)
            results["variants"][variant_name] = {
                "stats": self._compute_stats(variant_data),
                "comparison": comparison
            }
        
        return results
    
    def _compute_stats(self, data: List[float]) -> Dict:
        """Compute descriptive statistics."""
        if not data:
            return {}
        
        return {
            "count": len(data),
            "mean": statistics.mean(data),
            "median": statistics.median(data),
            "std": statistics.stdev(data) if len(data) > 1 else 0,
            "min": min(data),
            "max": max(data)
        }
    
    def _compare_groups(
        self,
        control: List[float],
        treatment: List[float]
    ) -> Dict:
        """Statistical comparison of two groups."""
        control_mean = statistics.mean(control)
        treatment_mean = statistics.mean(treatment)
        
        # Relative lift
        lift = (treatment_mean - control_mean) / control_mean if control_mean else 0
        
        # T-test
        t_stat, p_value = stats.ttest_ind(control, treatment)
        
        # Confidence interval (95%)
        diff = treatment_mean - control_mean
        se = (statistics.stdev(control)**2 / len(control) +
              statistics.stdev(treatment)**2 / len(treatment)) ** 0.5
        ci_low = diff - 1.96 * se
        ci_high = diff + 1.96 * se
        
        return {
            "lift_pct": lift * 100,
            "p_value": p_value,
            "significant": p_value < 0.05,
            "confidence_interval": (ci_low, ci_high),
            "t_statistic": t_stat
        }
    
    def required_sample_size(
        self,
        baseline_rate: float,
        mde: float,  # Minimum detectable effect
        alpha: float = 0.05,
        power: float = 0.8
    ) -> int:
        """
        Calculate required sample size per variant.
        
        Args:
            baseline_rate: Baseline conversion rate
            mde: Minimum detectable effect (relative)
            alpha: Significance level
            power: Statistical power
        
        Returns:
            Required samples per variant
        """
        p1 = baseline_rate
        p2 = baseline_rate * (1 + mde)
        
        pooled_p = (p1 + p2) / 2
        
        z_alpha = stats.norm.ppf(1 - alpha / 2)
        z_beta = stats.norm.ppf(power)
        
        n = (
            (z_alpha * (2 * pooled_p * (1 - pooled_p)) ** 0.5 +
             z_beta * (p1 * (1 - p1) + p2 * (1 - p2)) ** 0.5) ** 2
        ) / (p1 - p2) ** 2
        
        return int(n) + 1
```

    def assign_variant(self, user_id: str, experiment_name: str) -> Variant:
        """Assign user to variant using deterministic hashing."""
        exp = self.experiments[experiment_name]
        
        # Check cache
        if user_id in self.assignments:
            assigned_name = self.assignments[user_id].get(experiment_name)
            if assigned_name:
                for v in exp.variants:
                    if v.name == assigned_name:
                        return v
        
        # Hash-based assignment
        hash_input = f"{experiment_name}:{user_id}"
        hash_val = int(hashlib.md5(hash_input.encode()).hexdigest(), 16)
        bucket = (hash_val % 10000) / 10000
        
        cumulative = 0.0
        for variant in exp.variants:
            cumulative += variant.weight
            if bucket < cumulative:
                if user_id not in self.assignments:
                    self.assignments[user_id] = {}
                self.assignments[user_id][experiment_name] = variant.name
                return variant
        
        return exp.variants[-1]
    
    def log_metric(
        self,
        experiment_name: str,
        user_id: str,
        variant_name: str,
        metric_name: str,
        value: float
    ):
        """Log a metric for analysis."""
        self.metrics[experiment_name].append({
            "user_id": user_id,
            "variant": variant_name,
            "metric": metric_name,
            "value": value,
            "timestamp": datetime.utcnow()
        })
    
    def analyze_experiment(
        self,
        experiment_name: str,
        metric_name: str,
        control_variant: str = "control"
    ) -> Dict:
        """Analyze experiment results."""
        data = self.metrics[experiment_name]
        
        by_variant = {}
        for record in data:
            if record["metric"] == metric_name:
                v = record["variant"]
                by_variant.setdefault(v, []).append(record["value"])
        
        control_data = by_variant.get(control_variant, [])
        results = {
            "control_mean": statistics.mean(control_data) if control_data else 0,
            "variants": {}
        }
        
        for v_name, v_data in by_variant.items():
            if v_name == control_variant:
                continue
            
            v_mean = statistics.mean(v_data)
            lift = (v_mean - results["control_mean"]) / results["control_mean"] if results["control_mean"] else 0
            
            # T-test
            _, p_value = stats.ttest_ind(control_data, v_data)
            
            results["variants"][v_name] = {
                "mean": v_mean,
                "lift_pct": lift * 100,
                "p_value": p_value,
                "significant": p_value < 0.05
            }
        
        return results
