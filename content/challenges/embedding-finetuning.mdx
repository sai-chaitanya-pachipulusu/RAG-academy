# Embedding Model Fine-Tuning

Implement contrastive loss functions for domain-specific embedding fine-tuning, enabling significant retrieval quality improvements on specialized data.

## The Problem

General-purpose embeddings (text-embedding-3-small) are good, but **domain-specific embeddings** are better:
- Medical queries need to understand clinical terminology
- Legal documents require case law comprehension
- Code search needs programming language semantics

## Your Task

Implement core contrastive loss functions for embedding fine-tuning:

```python
# Training loop example
def train_step(model, batch, loss_fn, optimizer):
    """Single training step."""
    queries, positives, negatives = batch
    
    # Forward pass
    query_emb = model.encode(queries)
    positive_emb = model.encode(positives)
    negative_emb = model.encode(negatives) if negatives else None
    
    # Compute loss
    if loss_fn == "infonce":
        loss = losses.infonce_loss(query_emb, positive_emb, negative_emb)
    elif loss_fn == "triplet":
        loss = losses.triplet_loss(query_emb, positive_emb, negative_emb)
    elif loss_fn == "mnrl":
        loss = losses.multiple_negatives_ranking_loss(query_emb, positive_emb)
    
    # Backward pass
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    
    return loss.item()
```

## Algorithm Details

### InfoNCE Loss

```python
@staticmethod
def infonce_loss(query_emb: torch.Tensor,
                positive_emb: torch.Tensor,
                negative_embs: torch.Tensor,
                temperature: float = 0.07) -> torch.Tensor:
    """
    InfoNCE loss implementation.
    
    L = -log(exp(sim(q,p)/τ) / (exp(sim(q,p)/τ) + Σ exp(sim(q,n_i)/τ)))
    """
    # Normalize embeddings
    query_emb = F.normalize(query_emb, p=2, dim=-1)
    positive_emb = F.normalize(positive_emb, p=2, dim=-1)
    negative_embs = F.normalize(negative_embs, p=2, dim=-1)
    
    # Compute similarity with positive
    pos_sim = torch.sum(query_emb * positive_emb, dim=-1) / temperature
    
    # Compute similarities with negatives [batch_size, num_neg]
    neg_sim = torch.bmm(
        negative_embs,  # [batch, num_neg, dim]
        query_emb.unsqueeze(-1)  # [batch, dim, 1]
    ).squeeze(-1) / temperature
    
    # Concatenate positive and negative similarities
    # pos_sim: [batch], neg_sim: [batch, num_neg]
    logits = torch.cat([pos_sim.unsqueeze(1), neg_sim], dim=1)  # [batch, 1+num_neg]
    
    # Labels: positive is at index 0
    labels = torch.zeros(query_emb.size(0), dtype=torch.long, device=query_emb.device)
    
    # Cross-entropy loss
    loss = F.cross_entropy(logits, labels)
    
    return loss
```

### Triplet Loss

```python
@staticmethod
def triplet_loss(anchor_emb: torch.Tensor,
                positive_emb: torch.Tensor,
                negative_emb: torch.Tensor,
                margin: float = 0.5) -> torch.Tensor:
    """
    Triplet loss with margin.
    
    L = max(0, sim(a,n) - sim(a,p) + margin)
    """
    # Normalize embeddings
    anchor_emb = F.normalize(anchor_emb, p=2, dim=-1)
    positive_emb = F.normalize(positive_emb, p=2, dim=-1)
    negative_emb = F.normalize(negative_emb, p=2, dim=-1)
    
    # Compute similarities
    pos_sim = torch.sum(anchor_emb * positive_emb, dim=-1)
    neg_sim = torch.sum(anchor_emb * negative_emb, dim=-1)
    
    # Triplet loss
    loss = torch.clamp(neg_sim - pos_sim + margin, min=0.0)
    
    return loss.mean()
```

### Multiple Negatives Ranking Loss

```python
@staticmethod
def multiple_negatives_ranking_loss(query_emb: torch.Tensor,
                                    doc_emb: torch.Tensor,
                                    temperature: float = 0.07) -> torch.Tensor:
    """
    Multiple Negatives Ranking Loss.
    Uses other documents in the batch as negatives.
    
    For each query, the positive is at the same index in doc_emb.
    All other documents in the batch are treated as negatives.
    """
    batch_size = query_emb.size(0)
    
    # Normalize
    query_emb = F.normalize(query_emb, p=2, dim=-1)
    doc_emb = F.normalize(doc_emb, p=2, dim=-1)
    
    # Compute similarity matrix [batch_size, batch_size]
    # sim[i,j] = similarity between query i and document j
    sim_matrix = torch.matmul(query_emb, doc_emb.t()) / temperature
    
    # Labels: diagonal elements are positives
    labels = torch.arange(batch_size, device=query_emb.device)
    
    # Cross-entropy loss
    loss = F.cross_entropy(sim_matrix, labels)
    
    return loss
```

### Cosine Similarity

```python
@staticmethod
def cosine_similarity(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Compute cosine similarity between embeddings."""
    a_norm = F.normalize(a, p=2, dim=-1)
    b_norm = F.normalize(b, p=2, dim=-1)
    
    # Handle different shapes
    if a.dim() == 2 and b.dim() == 3:
        # a: [batch, dim], b: [batch, num_neg, dim]
        return torch.bmm(b_norm, a_norm.unsqueeze(-1)).squeeze(-1)
    else:
        return torch.sum(a_norm * b_norm, dim=-1)
```

## Loss Function Comparison

| Loss | Negatives | Memory | Quality | Use Case |
|------|-----------|--------|---------|----------|
| InfoNCE | External | High | Best | With pre-mined negatives |
| Triplet | Single | Low | Good | Simple fine-tuning |
| MNRL | In-batch | Medium | Very Good | Large batch training |

## Hard Negative Mining

```python
def mine_hard_negatives(query: str, positive: str, corpus: list[str], 
                       model, k: int = 10) -> list[str]:
    """
    Find documents similar to query but NOT the positive.
    
    Args:
        query: Query text
        positive: Positive document
        corpus: All documents to search
        model: Embedding model
        k: Number of hard negatives to return
        
    Returns:
        List of hard negative documents
    """
    # Encode query and corpus
    query_emb = model.encode(query)
    corpus_embs = model.encode(corpus)
    
    # Compute similarities
    similarities = np.dot(corpus_embs, query_emb)
    
    # Get top-k most similar
    top_k_idx = np.argsort(similarities)[-k:][::-1]
    
    # Filter out the positive
    hard_negatives = [
        corpus[i] for i in top_k_idx
        if corpus[i] != positive
    ]
    
    return hard_negatives[:k]
```
