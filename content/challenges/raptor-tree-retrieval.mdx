# RAPTOR Tree Retrieval

Implement efficient retrieval from RAPTOR hierarchical trees, enabling intelligent level selection and hierarchical navigation for both specific and broad queries.

## The Problem

Given a RAPTOR tree with multiple levels of summaries:
- **Which level** should we retrieve from?
- **How do we navigate** from summaries to details?
- **How do we combine** information across levels?

## Your Task

Implement a `RAPTORTreeRetriever` class:

## Algorithm Details

### Level Selection

```python
def select_level(self, query: str) -> int:
    """Select optimal level based on query characteristics."""
    query_lower = query.lower()
    
    # Heuristic-based selection
    broad_indicators = [
        "summarize", "overview", "themes", "main", "key points",
        "what are", "tell me about", "describe"
    ]
    
    specific_indicators = [
        "what was", "when did", "who", "how much", "specific",
        "exactly", "detail", "number", "date"
    ]
    
    broad_score = sum(1 for ind in broad_indicators if ind in query_lower)
    specific_score = sum(1 for ind in specific_indicators if ind in query_lower)
    
    if broad_score > specific_score:
        # Return middle-to-high level
        return max(1, len(self.tree["levels"]) // 2)
    else:
        # Return leaf level
        return 0
```

### Cross-Level Search

```python
def search_all_levels(self, query: str, k: int = 3) -> dict:
    """Search across all levels and return best matches per level."""
    query_embedding = self.embedder.encode([query])[0]
    
    level_results = {}
    for level_num, nodes in self.tree["levels"].items():
        embeddings = self.get_level_embeddings(level_num)
        
        # Calculate similarities
        similarities = cosine_similarity([query_embedding], embeddings)[0]
        
        # Get top-k for this level
        top_indices = np.argsort(similarities)[-k:][::-1]
        
        level_results[level_num] = [
            {
                "content": nodes[i],
                "score": similarities[i],
                "level": level_num,
                "id": f"L{level_num}_N{i}"
            }
            for i in top_indices
        ]
    
    return level_results
```

### Hierarchical Expansion

```python
def retrieve_with_expansion(self, query: str, k: int = 5,
                            expansion_depth: int = 1) -> list[dict]:
    """Retrieve and expand results hierarchically."""
    # Start at selected level
    start_level = self.select_level(query)
    candidates = self.search_level(query, start_level, k=k*2)
    
    expanded = []
    for node in candidates:
        expanded.append(node)
        
        # Expand to children if we want more detail
        if expansion_depth > 0 and start_level > 0:
            children = self.expand_node(node["id"], "children")
            for child in children[:3]:  # Limit expansion
                child["parent_score"] = node["score"]
                expanded.append(child)
        
        # Expand to parent if we want context
        if expansion_depth > 0 and start_level < max(self.tree["levels"].keys()):
            parent = self.expand_node(node["id"], "parent")
            if parent:
                parent[0]["child_score"] = node["score"]
                expanded.append(parent[0])
    
    # Rerank combined results
    return self.rerank(query, expanded, top_k=k)
```

### Multi-Level Fusion

```python
def multi_level_fusion(self, query: str, k_per_level: int = 3) -> list[dict]:
    """Combine insights from multiple levels."""
    all_results = []
    
    # Search each level
    for level in self.tree["levels"].keys():
        results = self.search_level(query, int(level), k=k_per_level)
        
        # Apply level weighting
        weight = self.level_weights.get(level, 1.0)
        for r in results:
            r["weighted_score"] = r["score"] * weight
        
        all_results.extend(results)
    
    # Deduplicate (same content may appear at multiple levels)
    seen_content = set()
    unique_results = []
    for r in sorted(all_results, key=lambda x: x["weighted_score"], reverse=True):
        content_hash = hash(r["content"][:100])  # First 100 chars
        if content_hash not in seen_content:
            seen_content.add(content_hash)
            unique_results.append(r)
    
    return unique_results[:k_per_level * 2]
```

## Navigation Strategies

### Strategy 1: Top-Down (Summary → Details)

```python
def top_down_retrieval(self, query: str, k: int = 5) -> list[dict]:
    """Start high, drill down to relevant branches."""
    # Start at root level
    current_level = max(self.tree["levels"].keys())
    candidates = self.search_level(query, current_level, k=3)
    
    results = []
    for node in candidates:
        if node["score"] > 0.8:  # High confidence
            # Drill down
            children = self.expand_node(node["id"], "children")
            child_matches = self.search_in_nodes(query, children, k=2)
            results.extend(child_matches)
        else:
            results.append(node)
    
    return results[:k]
```

### Strategy 2: Bottom-Up (Details → Context)

```python
def bottom_up_retrieval(self, query: str, k: int = 5) -> list[dict]:
    """Start at leaves, add parent context."""
    # Search leaves
    leaf_results = self.search_level(query, 0, k=k)
    
    results = []
    for node in leaf_results:
        # Add parent for context
        parent = self.expand_node(node["id"], "parent")
        if parent:
            results.append({
                "content": f"Context: {parent[0]['content']}\n\nDetail: {node['content']}",
                "score": node["score"],
                "sources": [parent[0]["id"], node["id"]]
            })
        else:
            results.append(node)
    
    return results
```

## Evaluation Criteria

1. **Level Selection Accuracy**: Chooses appropriate level for query type
2. **Retrieval Quality**: Returns relevant nodes
3. **Expansion Utility**: Children/parents add value
4. **Efficiency**: Reasonable query time

## Comparison: Retrieval Strategies

| Strategy | Best For | Latency | Coverage |
|----------|----------|---------|----------|
| Single Level | Known granularity | Low | Limited |
| Top-Down | Broad → Specific | Medium | Good |
| Bottom-Up | Specific + Context | Medium | Good |
| Multi-Level Fusion | Complex queries | Higher | Best |

## Production Considerations

1. **Index per level**: Build separate vector index for each level
2. **Lazy expansion**: Only expand high-confidence nodes
3. **Caching**: Cache level selection decisions
4. **Feedback loop**: Learn optimal levels from user clicks
