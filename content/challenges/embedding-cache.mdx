# Embedding Cache

Implement an embedding cache keyed by (model, text hash) to achieve 20-60% cost reduction on embedding API calls.

## Why This Matters in Production

In real systems, embedding calls are expensive and often repeated:

- Users ask the same question in different wording
- Your app re-embeds identical text by accident
- You re-index unchanged docs during ingestion

**Caching embeddings** provides:
- **Cheap to implement**: Simple hash-based lookup
- **Huge savings**: Often 20-60% fewer embed calls
- **Safer than semantic caching**: Simpler invalidation logic

## Offline vs Online

- **Offline (ingestion)**: Cache document/chunk embeddings so re-runs don't re-embed unchanged text
- **Online (query time)**: Cache query embeddings (same query text should not be re-embedded)

## Algorithm Details

### Cache Key Generation

```python
def _make_key(self, text: str, model: str) -> str:
    """Create deterministic cache key."""
    # Normalize text (optional)
    normalized = text.strip().lower()
    
    # Combine model and text
    key_content = f"{model}:{normalized}"
    
    # Hash for fixed-size key
    return hashlib.sha256(key_content.encode()).hexdigest()
```

### LRU Cache Implementation

```python
def __init__(self, max_size: int = 10000, eviction_policy: str = "lru"):
    self.max_size = max_size
    self.cache = OrderedDict()  # For LRU ordering
    self.stats = {"hits": 0, "misses": 0}

def get(self, text: str, model: str) -> Optional[list[float]]:
    key = self._make_key(text, model)
    
    if key in self.cache:
        # Move to end (most recently used)
        self.cache.move_to_end(key)
        self.stats["hits"] += 1
        return self.cache[key]["embedding"]
    
    self.stats["misses"] += 1
    return None

def set(self, text: str, model: str, embedding: list[float]) -> None:
    key = self._make_key(text, model)
    
    # Add to cache
    self.cache[key] = {
        "embedding": embedding,
        "text_hash": hashlib.sha256(text.encode()).hexdigest()[:16]
    }
    
    # Evict if over capacity
    if len(self.cache) > self.max_size:
        # Remove oldest (first item)
        self.cache.popitem(last=False)
```

### Cached Embedder

```python
class CachedEmbedder:
    def __init__(self, embedder, cache: EmbeddingCache):
        self.embedder = embedder
        self.cache = cache
    
    def embed(self, text: str, model: str = "default") -> list[float]:
        # Check cache
        cached = self.cache.get(text, model)
        if cached is not None:
            return cached
        
        # Cache miss - call embedder
        embedding = self.embedder.embed(text, model)
        
        # Store in cache
        self.cache.set(text, model, embedding)
        
        return embedding
    
    def embed_batch(self, texts: list[str], model: str = "default") -> list[list[float]]:
        results = []
        to_embed = []
        to_embed_indices = []
        
        # Check cache for each text
        for i, text in enumerate(texts):
            cached = self.cache.get(text, model)
            if cached is not None:
                results.append((i, cached))
            else:
                to_embed.append(text)
                to_embed_indices.append(i)
        
        # Batch embed uncached texts
        if to_embed:
            embeddings = self.embedder.embed_batch(to_embed, model)
            
            # Store in cache and add to results
            for idx, text, emb in zip(to_embed_indices, to_embed, embeddings):
                self.cache.set(text, model, emb)
                results.append((idx, emb))
        
        # Sort by original index
        results.sort(key=lambda x: x[0])
        return [emb for _, emb in results]
```

## Production Extensions

### Redis Backend

```python
import redis
import json

class RedisEmbeddingCache(EmbeddingCache):
    """Distributed embedding cache using Redis."""
    
    def __init__(self, redis_client: redis.Redis, ttl_seconds: int = 86400):
        self.redis = redis_client
        self.ttl = ttl_seconds
    
    def get(self, text: str, model: str) -> Optional[list[float]]:
        key = self._make_key(text, model)
        cached = self.redis.get(key)
        
        if cached:
            return json.loads(cached)
        return None
    
    def set(self, text: str, model: str, embedding: list[float]) -> None:
        key = self._make_key(text, model)
        self.redis.setex(key, self.ttl, json.dumps(embedding))
```

### Versioned Cache Keys

```python
def _make_key(self, text: str, model: str, version: str = "v1") -> str:
    """Include version for cache invalidation on model updates."""
    key_content = f"{model}:{version}:{text.strip().lower()}"
    return hashlib.sha256(key_content.encode()).hexdigest()
```

## Evaluation Criteria

1. **Correctness**: Same (model, text) returns cached embedding
2. **Eviction**: LRU/LFU eviction when capacity reached
3. **Statistics**: Accurate hit/miss tracking
4. **Batch Efficiency**: Deduplicates within batch

## Test Cases

```python
def test_cache_hit():
    """Test that identical queries hit cache."""
    cache = EmbeddingCache()
    
    text = "test query"
    model = "test-model"
    embedding = [0.1, 0.2, 0.3]
    
    cache.set(text, model, embedding)
    result = cache.get(text, model)
    
    assert result == embedding
    assert cache.get_stats()["hits"] == 1

def test_model_specific():
    """Test that different models have different cache entries."""
    cache = EmbeddingCache()
    
    text = "test"
    cache.set(text, "model-a", [1.0, 2.0])
    cache.set(text, "model-b", [3.0, 4.0])
    
    assert cache.get(text, "model-a") == [1.0, 2.0]
    assert cache.get(text, "model-b") == [3.0, 4.0]

def test_lru_eviction():
    """Test LRU eviction policy."""
    cache = EmbeddingCache(max_size=2)
    
    cache.set("a", "model", [1.0])
    cache.set("b", "model", [2.0])
    cache.get("a", "model")  # Access "a" to make it recently used
    cache.set("c", "model", [3.0])  # Should evict "b"
    
    assert cache.get("a", "model") is not None
    assert cache.get("b", "model") is None  # Evicted
    assert cache.get("c", "model") is not None
```
