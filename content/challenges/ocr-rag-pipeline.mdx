# OCR + RAG Pipeline

Integrate OCR text extraction into your RAG pipeline to process scanned documents, PDF images, and visual text content.

## The Problem

Many documents contain text as images:
- **Scanned contracts** and legal documents
- **PDFs** that are just image scans
- **Handwritten notes** and forms
- **Screenshots** and embedded images

Without OCR, this content is invisible to RAG systems.

## Your Task

Implement an `OCRPipeline` class:


## Algorithm Details

### OCR with PaddleOCR

```python
def extract_text(self, image: Union[str, Image.Image]) -> dict:
    """Extract text using PaddleOCR."""
    from paddleocr import PaddleOCR
    
    if self.ocr is None:
        self.ocr = PaddleOCR(
            use_angle_cls=True,
            lang=self.language,
            show_log=False
        )
    
    # Load image
    if isinstance(image, str):
        img_path = image
    else:
        # Save temp file
        import tempfile
        with tempfile.NamedTemporaryFile(suffix='.png', delete=False) as f:
            image.save(f.name)
            img_path = f.name
    
    # Run OCR
    result = self.ocr.ocr(img_path, cls=True)
    
    # Parse results
    blocks = []
    all_text = []
    total_confidence = 0
    num_blocks = 0
    
    for line in result[0]:
        if line:
            bbox = line[0]  # Bounding box
            text = line[1][0]  # Text content
            confidence = line[1][1]  # Confidence score
            
            blocks.append({
                "text": text,
                "confidence": confidence,
                "bbox": bbox,
                "page": 1
            })
            
            all_text.append(text)
            total_confidence += confidence
            num_blocks += 1
    
    full_text = " ".join(all_text)
    avg_confidence = total_confidence / num_blocks if num_blocks > 0 else 0
    
    return {
        "text": full_text,
        "confidence": avg_confidence,
        "blocks": blocks,
        "word_count": len(full_text.split())
    }
```

### Quality Checking

```python
def quality_check(self, extraction: dict) -> dict:
    """Validate OCR extraction quality."""
    issues = []
    suggestions = []
    
    # Check overall confidence
    if extraction["confidence"] < self.confidence_threshold:
        issues.append(f"Low confidence: {extraction['confidence']:.2%}")
        suggestions.append("Consider image preprocessing (denoising, deskewing)")
    
    # Check for garbage text patterns
    text = extraction["text"]
    garbage_patterns = [r'[^\w\s]{5,}', r'\d{10,}']  # 5+ special chars, 10+ digits
    for pattern in garbage_patterns:
        if re.search(pattern, text):
            issues.append("Possible garbage text detected")
            suggestions.append("Review and clean extracted text manually")
            break
    
    # Check word count
    if extraction["word_count"] < 5:
        issues.append("Very short extraction")
        suggestions.append("Verify image contains text")
    
    # Check block confidence distribution
    low_conf_blocks = [b for b in extraction["blocks"] 
                      if b["confidence"] < 0.5]
    if len(low_conf_blocks) > len(extraction["blocks"]) * 0.3:
        issues.append(f"{len(low_conf_blocks)} blocks have low confidence")
        suggestions.append("Consider using a different OCR engine")
    
    return {
        "passed": len(issues) == 0,
        "issues": issues,
        "suggestions": suggestions
    }
```

### PDF Processing

```python
def process_pdf(self, pdf_path: str, 
               extract_images: bool = True) -> list[dict]:
    """Process PDF with OCR for image-based pages."""
    from pdf2image import convert_from_path
    import fitz  # PyMuPDF
    
    results = []
    
    # Convert PDF to images
    doc = fitz.open(pdf_path)
    
    for page_num in range(len(doc)):
        page = doc[page_num]
        
        # First try to extract text directly
        text = page.get_text()
        
        if text.strip():
            # Page has extractable text
            results.append({
                "page": page_num + 1,
                "text": text,
                "source": "native",
                "confidence": 1.0
            })
        else:
            # Page is image-based, use OCR
            pix = page.get_pixmap(matrix=fitz.Matrix(2, 2))  # 2x zoom for better OCR
            img = Image.frombytes("RGB", [pix.width, pix.height], pix.samples)
            
            ocr_result = self.extract_text(img)
            ocr_result["page"] = page_num + 1
            ocr_result["source"] = "ocr"
            results.append(ocr_result)
    
    doc.close()
    return results
```

## Evaluation Criteria

1. **Text Accuracy**: High character/word accuracy
2. **Confidence Scoring**: Reliable confidence estimates
3. **Layout Preservation**: Correct reading order
4. **Quality Gating**: Rejects poor extractions

## Integration with RAG

```python
class OCRRAGPipeline:
    def __init__(self):
        self.ocr = OCRPipeline()
        self.embedder = TextEmbedder()
        self.vector_store = VectorStore()
    
    def index_document(self, file_path: str, metadata: dict = None):
        """Index document with OCR support."""
        if file_path.endswith('.pdf'):
            pages = self.ocr.process_pdf(file_path)
        else:
            extraction = self.ocr.extract_text(file_path)
            pages = [extraction]
        
        for page in pages:
            # Quality filter
            quality = self.ocr.quality_check(page)
            if not quality["passed"]:
                print(f"Skipping page {page.get('page', 0)}: {quality['issues']}")
                continue
            
            # Chunk and index
            chunks = self.ocr.to_chunks(page)
            for chunk in chunks:
                embedding = self.embedder.embed(chunk["text"])
                self.vector_store.add(embedding, {
                    "text": chunk["text"],
                    "source": file_path,
                    "page": page.get("page", 1),
                    "ocr_confidence": page["confidence"],
                    **(metadata or {})
                })
```
