## The "Overkill" Problem

Not every query needs a flagship model like GPT-4o. Simple tasks like "Summarize my previous message" or "What is my account balance?" can be handled perfectly by cheaper, faster models (e.g., GPT-4o-mini).

**Cost-Aware Routing** involves:

1.  Analyzing query complexity (length, intent, reasoning type).
2.  Routing to a "Small" model for simple tasks.
3.  Routing to a "Large" model for complex tasks.

## Implementation

```python
from typing import Callable, Dict, List, Optional
from dataclasses import dataclass
import re

@dataclass
class ModelConfig:
    """Configuration for a model tier."""
    name: str
    cost_per_1k_input: float
    cost_per_1k_output: float
    avg_latency_ms: float
    model_fn: Callable[[str], str]


class CostAwareRouter:
    """
    Router that selects models based on query complexity to optimize cost.
    
    Routes simple queries to cheaper models, complex queries to powerful models.
    """
    
    def __init__(
        self,
        small_model: ModelConfig,
        medium_model: ModelConfig,
        large_model: ModelConfig
    ):
        """
        Initialize with three model tiers.
        
        Args:
            small_model: Cheap, fast model (e.g., GPT-4o-mini)
            medium_model: Balanced model (e.g., GPT-4o)
            large_model: Powerful model (e.g., Claude 3 Opus)
        """
        self.small_model = small_model
        self.medium_model = medium_model
        self.large_model = large_model
        
        # Complexity patterns
        self.simple_patterns = [
            r'^\s*\d+\s*[+\-*/]\s*\d+',  # Math
            r'^(hi|hello|hey)\b',  # Greetings
            r'\b(yes|no|true|false)\b',  # Boolean
        ]
        
        self.complex_patterns = [
            r'\b(compare|analyze|explain|why|how does)\b',
            r'\b(design|implement|create|write)\s+(a|an|the)\s+\w+',
            r'\b(code|function|algorithm|architecture)\b',
        ]
    
    def estimate_complexity(self, query: str) -> str:
        """
        Estimate query complexity.
        
        Returns:
            'simple', 'medium', or 'complex'
        """
        query_lower = query.lower().strip()
        word_count = len(query_lower.split())
        
        # Check simple patterns
        for pattern in self.simple_patterns:
            if re.search(pattern, query_lower):
                return 'simple'
        
        # Check complex patterns
        for pattern in self.complex_patterns:
            if re.search(pattern, query_lower):
                return 'complex'
        
        # Length-based heuristics
        if word_count <= 5:
            return 'simple'
        elif word_count >= 20:
            return 'complex'
        
        return 'medium'
    
    def select_model(self, query: str) -> ModelConfig:
        """
        Select appropriate model for query.
        
        Args:
            query: User query
        
        Returns:
            Selected model configuration
        """
        complexity = self.estimate_complexity(query)
        
        if complexity == 'simple':
            return self.small_model
        elif complexity == 'complex':
            return self.large_model
        else:
            return self.medium_model
    
    def route(self, query: str) -> Dict:
        """
        Route query to selected model and return result with cost info.
        
        Args:
            query: User query
        
        Returns:
            Dict with answer, model used, and estimated cost
        """
        model = self.select_model(query)
        
        # Call model
        answer = model.model_fn(query)
        
        # Estimate cost (simplified)
        input_tokens = len(query.split()) * 1.3  # Rough estimate
        output_tokens = len(answer.split()) * 1.3
        
        input_cost = (input_tokens / 1000) * model.cost_per_1k_input
        output_cost = (output_tokens / 1000) * model.cost_per_1k_output
        total_cost = input_cost + output_cost
        
        return {
            'answer': answer,
            'model': model.name,
            'complexity': self.estimate_complexity(query),
            'estimated_cost_usd': total_cost,
            'estimated_latency_ms': model.avg_latency_ms
        }
    
    def compare_costs(self, query: str) -> Dict:
        """
        Show what each model tier would cost for this query.
        
        Useful for debugging and cost analysis.
        """
        input_tokens = len(query.split()) * 1.3
        
        results = {}
        for model in [self.small_model, self.medium_model, self.large_model]:
            # Assume similar output size for comparison
            output_tokens = input_tokens * 2
            
            input_cost = (input_tokens / 1000) * model.cost_per_1k_input
            output_cost = (output_tokens / 1000) * model.cost_per_1k_output
            
            results[model.name] = {
                'input_cost': input_cost,
                'output_cost': output_cost,
                'total_cost': input_cost + output_cost,
                'latency_ms': model.avg_latency_ms
            }
        
        return results
```

## Best Practices

1. **Monitor accuracy**: Ensure cheap models maintain quality
2. **Fallback gracefully**: Route to better model if cheap one fails
3. **Track savings**: Measure actual cost reduction
4. **A/B test**: Compare routing vs single model
5. **User feedback**: Learn from corrections
