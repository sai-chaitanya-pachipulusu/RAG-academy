## When to Fine-tune Embeddings

**Fine-tune when:**

- Domain-specific vocabulary (legal, medical, code)
- Base model underperforms on your data
- You have labeled pairs or synthetic data
- 5-10% quality improvement justifies the effort

**Skip fine-tuning when:**

- General domain works well
- No training data available
- Quick iteration is priority
- Base model quality is sufficient

## Fine-tuning Approaches

### 1. Contrastive Learning (Most Common)

Train embeddings to bring similar pairs closer, push dissimilar apart:

```python
from sentence_transformers import SentenceTransformer, losses
from sentence_transformers import InputExample
from torch.utils.data import DataLoader

# Load base model
model = SentenceTransformer('BAAI/bge-base-en-v1.5')

# Prepare training data
train_examples = [
    InputExample(texts=["query 1", "relevant doc 1"]),
    InputExample(texts=["query 2", "relevant doc 2"]),
    # ...
]

train_dataloader = DataLoader(train_examples, shuffle=True, batch_size=16)

# Define loss
train_loss = losses.MultipleNegativesRankingLoss(model)

# Train
model.fit(
    train_objectives=[(train_dataloader, train_loss)],
    epochs=3,
    warmup_steps=100,
    output_path="./finetuned-embeddings"
)
```

### 2. With Hard Negatives (Better Quality)

Hard negatives are documents that seem similar but aren't relevant:

```python
# Mine hard negatives
def mine_hard_negatives(query: str, positive: str, corpus: list, k: int = 10):
    """Find documents similar to query but NOT the positive."""
    query_emb = model.encode(query)
    corpus_embs = model.encode(corpus)

    similarities = query_emb @ corpus_embs.T
    top_k_idx = similarities.argsort()[-k:][::-1]

    hard_negatives = [
        corpus[i] for i in top_k_idx
        if corpus[i] != positive
    ]
    return hard_negatives

# Training with hard negatives
train_examples = []
for query, positive in pairs:
    hard_negs = mine_hard_negatives(query, positive, corpus)

    for neg in hard_negs[:3]:
        train_examples.append(
            InputExample(texts=[query, positive, neg])
        )

# Use TripletLoss
train_loss = losses.TripletLoss(model)
```

### 3. Matryoshka Training (Flexible Dimensions)

Train embeddings that work at multiple dimensions:

```python
from sentence_transformers import SentenceTransformer
from sentence_transformers.losses import MatryoshkaLoss, MultipleNegativesRankingLoss

model = SentenceTransformer("BAAI/bge-base-en-v1.5")

# Wrap with Matryoshka
base_loss = MultipleNegativesRankingLoss(model)
loss = MatryoshkaLoss(
    model,
    base_loss,
    matryoshka_dims=[768, 512, 256, 128, 64]
)

# Now the model works at any of these dimensions
model.fit(train_objectives=[(train_dataloader, loss)], epochs=3)

# Use at different dimensions
emb_768 = model.encode("query", output_value="sentence_embedding")[:768]
emb_256 = model.encode("query", output_value="sentence_embedding")[:256]
```

## Data Preparation

### From Existing Retrieval Logs

```python
def create_training_data_from_logs(logs: list[dict]) -> list[tuple]:
    """
    Extract positive pairs from user interactions.
    Logs: [{query, clicked_doc, results}, ...]
    """
    pairs = []

    for log in logs:
        query = log["query"]

        # Clicked doc = positive
        if log.get("clicked_doc"):
            pairs.append((query, log["clicked_doc"]))

        # High-rated answer = positive
        if log.get("rating", 0) >= 4:
            pairs.append((query, log["retrieved_docs"][0]))

    return pairs
```

### Synthetic Data Generation

```python
def generate_synthetic_queries(documents: list[str], num_per_doc: int = 3) -> list[tuple]:
```

### From Q&A Datasets

```python
def prepare_qa_pairs(qa_data: list[dict]) -> list[tuple]:
    """Convert Q&A data to training pairs."""
    pairs = []

    for item in qa_data:
        question = item["question"]
        # Use the context/answer as positive
        context = item.get("context", item.get("answer", ""))

        if question and context:
            pairs.append((question, context))

    return pairs
```

## Training Best Practices

### 1. Data Quality > Quantity

```python
# Filter low-quality pairs
def filter_training_data(pairs: list[tuple], min_query_len: int = 10, min_doc_len: int = 50):
    filtered = []

    for query, doc in pairs:
        if len(query) >= min_query_len and len(doc) >= min_doc_len:
            # Check for actual relevance (optional LLM check)
            filtered.append((query, doc))

    return filtered
```

### 2. Balanced Batches

```python
# Ensure diversity in batches
from sentence_transformers import SentenceTransformer
from sentence_transformers.datasets import NoDuplicatesDataLoader

# This loader ensures no duplicate queries in same batch
train_dataloader = NoDuplicatesDataLoader(
    train_examples,
    batch_size=32
)
```

### 3. Evaluation During Training

```python
from sentence_transformers.evaluation import InformationRetrievalEvaluator

# Create evaluation set
ir_evaluator = InformationRetrievalEvaluator(
    queries={"q1": "What is X?", "q2": "How does Y work?"},
    corpus={"d1": "X is...", "d2": "Y works by...", "d3": "..."},
    relevant_docs={"q1": ["d1"], "q2": ["d2"]},
    name="dev-eval"
)

model.fit(
    train_objectives=[(train_dataloader, train_loss)],
    evaluator=ir_evaluator,
    evaluation_steps=500,
    epochs=3
)
```

## LoRA Fine-tuning (Memory Efficient)

For large models, use LoRA:

```python
from peft import LoraConfig, get_peft_model
from transformers import AutoModel

# Load base model
base_model = AutoModel.from_pretrained("BAAI/bge-large-en-v1.5")

# Configure LoRA
lora_config = LoraConfig(
    r=8,                      # Rank
    lora_alpha=16,
    target_modules=["query", "key", "value"],
    lora_dropout=0.1,
)

# Wrap with LoRA
model = get_peft_model(base_model, lora_config)

# Only ~0.1% of parameters are trainable now
trainable_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
total_params = sum(p.numel() for p in model.parameters())
print(f"Trainable: {trainable_params/total_params:.2%}")
```

## Evaluation Metrics

```python
def evaluate_embeddings(model, test_pairs: list[tuple], corpus: list[str]) -> dict:
    """Evaluate embedding model on retrieval task."""
    queries = [p[0] for p in test_pairs]
    positives = [p[1] for p in test_pairs]

    # Encode all
    query_embs = model.encode(queries)
    corpus_embs = model.encode(corpus)

    # Compute retrieval metrics
    recall_at_1 = 0
    recall_at_10 = 0
    mrr = 0

    for i, query_emb in enumerate(query_embs):
        similarities = query_emb @ corpus_embs.T
        ranking = similarities.argsort()[::-1]

        positive_idx = corpus.index(positives[i])
        rank = list(ranking).index(positive_idx) + 1

        if rank <= 1:
            recall_at_1 += 1
        if rank <= 10:
            recall_at_10 += 1
        mrr += 1 / rank

    n = len(test_pairs)
    return {
        "recall@1": recall_at_1 / n,
        "recall@10": recall_at_10 / n,
        "mrr": mrr / n,
    }
```

## Cost Analysis

| Approach            | Data Needed  | Compute        | Time    |
| ------------------- | ------------ | -------------- | ------- |
| Contrastive (small) | 10K pairs    | 1 GPU-hour     | 1 day   |
| Hard negatives      | 50K triplets | 4 GPU-hours    | 2 days  |
| LoRA (large model)  | 100K pairs   | 8 GPU-hours    | 3 days  |
| Full fine-tune      | 500K+ pairs  | 100+ GPU-hours | 1+ week |
