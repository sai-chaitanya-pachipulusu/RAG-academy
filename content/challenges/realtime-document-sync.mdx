# Real-Time Document Sync

Traditional RAG systems re-index documents in batches, leading to stale knowledge. Production systems need real-time updates.

**Your task**: Build an event-driven system that keeps the vector index in sync with document changes.

## Setup

```python
class MockVectorStore:

        self.operations: List[str] = []  # Track operations for testing
    
    async def upsert(self, chunks: List[Chunk]):
        for chunk in chunks:
            self.vectors[chunk.id] = chunk
            self.operations.append(f"UPSERT {chunk.id}")
    
    async def delete(self, filter: Dict):
        doc_id = filter.get("doc_id")
        to_delete = [k for k, v in self.vectors.items() if v.doc_id == doc_id]
        for k in to_delete:
            del self.vectors[k]
            self.operations.append(f"DELETE {k}")
    
    async def query(self, embedding: List[float], k: int = 5) -> List[Chunk]:
        # Simple mock: return first k chunks
        return list(self.vectors.values())[:k]

# Simulated embedding function
def mock_embed(texts: List[str]) -> List[List[float]]:
    """Generate deterministic fake embeddings."""
    return [[hash(t) % 1000 / 1000.0] * 128 for t in texts]
```

## Requirements

1. **Idempotent updates**: Processing same event twice has same result
2. **Change detection**: Skip re-indexing if content unchanged
3. **Batch efficiency**: Process multiple events together when possible
4. **Proper deletion**: Remove all chunks when document deleted

<summary>Hint 1: Stable chunk IDs</summary>

Use content hash for stable, reproducible chunk IDs:

```python
def chunk_document(self, doc_id: str, content: str) -> List[Chunk]:
    chunks = []
    
    # Simple chunking by character count
    for i in range(0, len(content), self.chunk_size):
        chunk_text = content[i:i + self.chunk_size]
        
        # Stable ID based on position and content
        chunk_id = f"{doc_id}_chunk_{i//self.chunk_size}"
        
        chunks.append(Chunk(
            id=chunk_id,
            doc_id=doc_id,
            text=chunk_text,
            embedding=mock_embed([chunk_text])[0],
            metadata={"position": i}
        ))
    
    return chunks
```

<details>
<summary>Hint 2: Change detection</summary>

Skip processing if content hash matches:

```python
async def process_event(self, event: DocumentEvent) -> Dict:
    if event.type == "deleted":
        await self.vector_store.delete({"doc_id": event.doc_id})
        self.doc_versions.pop(event.doc_id, None)
        return {"action": "deleted", "chunks_affected": 1, "skipped": False}
    
    # Check for unchanged content
    content_hash = self.compute_content_hash(event.content)
    if self.doc_versions.get(event.doc_id) == content_hash:
        return {"action": "skipped", "chunks_affected": 0, "skipped": True}
    
    # Delete old chunks, add new ones
    await self.vector_store.delete({"doc_id": event.doc_id})
    chunks = self.chunk_document(event.doc_id, event.content)
    await self.vector_store.upsert(chunks)
    
    self.doc_versions[event.doc_id] = content_hash
    return {"action": event.type, "chunks_affected": len(chunks), "skipped": False}
```

</details>

<summary>Hint 3: Batch processing</summary>

Deduplicate and process efficiently:

```python
async def process_batch(self, events: List[DocumentEvent]) -> List[Dict]:
    # Keep only latest event per doc_id
    latest_by_doc: Dict[str, DocumentEvent] = {}
    for event in events:
        existing = latest_by_doc.get(event.doc_id)
        if not existing or event.timestamp > existing.timestamp:
            latest_by_doc[event.doc_id] = event
    
    # Process deduplicated events
    results = []
    for event in latest_by_doc.values():
        result = await self.process_event(event)
        results.append(result)
    
    return results
```
