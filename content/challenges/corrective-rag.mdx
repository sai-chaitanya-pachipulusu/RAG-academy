# Corrective RAG (CRAG) Evaluator

Retrieved documents aren't always relevant. CRAG adds a **grading step** before generation to ensure quality.

## The CRAG Pipeline

```
Query → Retrieve Top-K → Grade Documents → Decision
                                              │
                    ┌─────────────────────────┼─────────────────────────┐
                    ↓                         ↓                         ↓
              All Relevant              Some Relevant             All Irrelevant
                    │                         │                         │
                    ↓                         ↓                         ↓
            Use All Docs           Filter + Use Relevant      Trigger Fallback
                                                                  │
                                                                  ├─→ Web Search
                                                                  ├─→ LLM Generate
                                                                  └─→ Ask Clarification
```

## Document Grades

| Grade      | Action                  | Confidence |
| ---------- | ----------------------- | ---------- |
| Relevant   | Use in context          | >0.8       |
| Ambiguous  | Maybe use, lower weight | 0.5-0.8    |
| Irrelevant | Discard                 | <0.5       |

## Implementation

### Document Grader

```python
from dataclasses import dataclass
from typing import List, Dict, Literal
from enum import Enum

class DocumentGrade(Enum):
    RELEVANT = "relevant"
    AMBIGUOUS = "ambiguous"
    IRRELEVANT = "irrelevant"

@dataclass
class GradedDocument:
    doc_id: str
    text: str
    grade: DocumentGrade
    confidence: float
    reasoning: str

class DocumentGrader:
    """
    Grade documents for relevance to a query.

    Uses LLM-based grading with confidence scoring.
    """

    GRADING_PROMPT = """Grade the relevance of this document to the query.

Query: {query}

Document:
{document}

Grade the document:
- RELEVANT: Directly answers or highly relevant to the query
- AMBIGUOUS: Somewhat related but not clearly helpful
- IRRELEVANT: Not useful for answering the query

Also provide:
- Confidence score (0.0-1.0)
- Brief reasoning

Format your response as:
Grade: RELEVANT/AMBIGUOUS/IRRELEVANT
Confidence: 0.XX
Reasoning: Your explanation"""

    def __init__(self, llm_client):
        self.llm = llm_client

    def grade_document(self, query: str, document: str) -> GradedDocument:
        """Grade a single document."""
        prompt = self.GRADING_PROMPT.format(
            query=query,
            document=document[:2000]  # Truncate long docs
        )

        response = self.llm.generate(prompt, max_tokens=150)

        # Parse response
        grade, confidence, reasoning = self._parse_response(response)

        return GradedDocument(
            doc_id="doc",  # Will be set by caller
            text=document,
            grade=grade,
            confidence=confidence,
            reasoning=reasoning
        )

    def grade_documents(
        self,
        query: str,
        documents: List[Dict]
    ) -> List[GradedDocument]:
        """Grade multiple documents."""
        graded = []

        for doc in documents:
            graded_doc = self.grade_document(query, doc["text"])
            graded_doc.doc_id = doc.get("id", "unknown")
            graded.append(graded_doc)

        return graded

    def _parse_response(self, response: str) -> tuple:
        """Parse grading response."""
        lines = response.strip().split("\n")

        grade = DocumentGrade.AMBIGUOUS  # Default
        confidence = 0.5
        reasoning = ""

        for line in lines:
            line_lower = line.lower()

            if line_lower.startswith("grade:"):
                grade_str = line.split(":", 1)[1].strip().upper()
                if "RELEVANT" in grade_str and "IR" not in grade_str:
                    grade = DocumentGrade.RELEVANT
                elif "IRRELEVANT" in grade_str:
                    grade = DocumentGrade.IRRELEVANT
                else:
                    grade = DocumentGrade.AMBIGUOUS

            elif line_lower.startswith("confidence:"):
                try:
                    confidence = float(line.split(":", 1)[1].strip())
                except:
                    pass

            elif line_lower.startswith("reasoning:"):
                reasoning = line.split(":", 1)[1].strip()

        return grade, confidence, reasoning
```

import re

class HeuristicGrader:
    """Fast heuristic-based document grading."""

    def __init__(self):
        self.min_keyword_overlap = 2
        self.min_semantic_score = 0.3

    def grade_document(
        self,
        query: str,
        document: str
    ) -> GradedDocument:
        """
        Grade document using heuristics.

        Fast but less accurate than LLM grading.
        Use for initial filtering.
        """
        # Extract keywords from query
        query_keywords = self._extract_keywords(query)
        doc_keywords = self._extract_keywords(document)

        # Calculate overlap
        overlap = len(set(query_keywords) & set(doc_keywords))
        overlap_ratio = overlap / max(len(query_keywords), 1)

        # Calculate semantic similarity (if embeddings available)
        semantic_score = self._semantic_similarity(query, document)

        # Combined score
        combined_score = 0.6 * overlap_ratio + 0.4 * semantic_score

        # Grade based on score
        if combined_score >= 0.7:
            grade = DocumentGrade.RELEVANT
        elif combined_score >= 0.4:
            grade = DocumentGrade.AMBIGUOUS
        else:
            grade = DocumentGrade.IRRELEVANT

        return GradedDocument(
            doc_id="doc",
            text=document,
            grade=grade,
            confidence=combined_score,

### CRAG Pipeline

```python
class CorrectiveRAG:
    """RAG with corrective document grading and fallbacks."""

    def __init__(
        self,
        retriever,
        llm,
        grader: DocumentGrader,
        web_search=None,
        relevance_threshold: float = 0.5
    ):
        self.retriever = retriever
        self.llm = llm
        self.grader = grader
        self.web_search = web_search
        self.relevance_threshold = relevance_threshold

    def query(self, question: str) -> Dict:
        """Execute CRAG pipeline."""
        # Step 1: Retrieve
        docs = self.retriever.search(question, k=5)

        if not docs:
            return self._handle_no_documents(question)

        # Step 2: Grade documents
        graded = self.grader.grade_documents(question, docs)

        # Step 3: Categorize
        relevant = [g for g in graded if g.grade == DocumentGrade.RELEVANT]
        ambiguous = [g for g in graded if g.grade == DocumentGrade.AMBIGUOUS]
        irrelevant = [g for g in graded if g.grade == DocumentGrade.IRRELEVANT]

        # Step 4: Decide action
        if relevant:
            # Use relevant documents
            context = self._format_context(relevant)
            answer = self._generate(question, context)

            return {
                "answer": answer,
                "source": "retrieval",
                "documents_used": [r.doc_id for r in relevant],
                "grades": {g.doc_id: g.grade.value for g in graded}
            }

        elif ambiguous and not self.web_search:
            # Use ambiguous with warning
            context = self._format_context(ambiguous)
            answer = self._generate_with_caveat(question, context)

            return {
                "answer": answer,
                "source": "retrieval_ambiguous",
                "documents_used": [a.doc_id for a in ambiguous],
                "warning": "Documents may not fully answer the question"
            }

        elif self.web_search:
            # Fallback to web search
            return self._web_fallback(question)

        else:
            # Generate without context
            return self._generate_without_context(question)

    def _format_context(self, graded_docs: List[GradedDocument]) -> str:
        """Format graded documents for context."""
        parts = []
        for doc in graded_docs:
            parts.append(f"[Relevance: {doc.confidence:.2f}] {doc.text}")
        return "\n\n".join(parts)

    def _generate(self, question: str, context: str) -> str:
        """Generate answer with context."""
        prompt = f"""Answer the question based on the context provided.

Context:
{context}

Question: {question}

Answer:"""

        return self.llm.generate(prompt)

    def _generate_with_caveat(self, question: str, context: str) -> str:
        """Generate with warning about document quality."""
        prompt = f"""The following documents may not fully answer the question.
Answer based on what information is available, and note any gaps.

Context:
{context}

Question: {question}

Answer:"""

        return self.llm.generate(prompt)

    def _generate_without_context(self, question: str) -> str:
        """Generate without retrieval context."""
        prompt = f"""No relevant documents were found for this question.
Answer based on your knowledge, or indicate if you cannot answer.

Question: {question}

Answer:"""

        return self.llm.generate(prompt)

    def _web_fallback(self, question: str) -> Dict:
        """Fallback to web search."""
        if not self.web_search:
            return self._generate_without_context(question)

        web_results = self.web_search.search(question)

        if not web_results:
            return self._generate_without_context(question)

        context = "\n\n".join([r["snippet"] for r in web_results[:3]])
        answer = self._generate(question, context)

        return {
            "answer": answer,
            "source": "web_fallback",
            "web_sources": [r["url"] for r in web_results[:3]]
        }

    def _handle_no_documents(self, question: str) -> Dict:
        """Handle case when no documents retrieved."""
        if self.web_search:
            return self._web_fallback(question)

        answer = self._generate_without_context(question)

        return {
            "answer": answer,
            "source": "no_documents",
            "warning": "No documents found in knowledge base"
        }
```

## Decision Matrix

| Relevant | Ambiguous | Irrelevant | Action |
|----------|-----------|------------|--------|
| ≥1 | any | any | Use relevant docs |
| 0 | ≥1 | any | Use ambiguous with caveat |
| 0 | 0 | all | Trigger fallback |

## Best Practices

1. **Parallel grading**: Grade all docs simultaneously
2. **Confidence thresholds**: Tune based on your use case
3. **Fallback strategy**: Always have a backup plan
4. **Logging**: Track grading decisions for improvement
5. **Hybrid approach**: Use heuristics first, LLM for edge cases
