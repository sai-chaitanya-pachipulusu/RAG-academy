# Multi-Step Reasoning Agent

Build an agent that breaks complex queries into sub-steps and executes them sequentially.

## Why This Matters

Single-shot RAG fails on complex questions. Multi-step reasoning enables "chain of thought" retrieval for multi-hop queries.

| Query Type | Single-Shot | Multi-Step |
|------------|-------------|------------|
| "What is X?" | ✅ Works | ✅ Works |
| "Compare X and Y" | ⚠️ Partial | ✅ Works |
| "What caused X to affect Y?" | ❌ Fails | ✅ Works |
| "Summarize trends from A, B, C" | ❌ Fails | ✅ Works |

- Sequential execution
- Result synthesis

### Reasoning Plan

```python
from dataclasses import dataclass
from typing import List, Optional, Literal
from enum import Enum

class StepType(Enum):
    RETRIEVE = "retrieve"      # Fetch relevant documents
    REASON = "reason"          # Generate answer from context
    CALCULATE = "calculate"    # Perform computation
    SYNTHESIZE = "synthesize"  # Combine multiple results
    ANSWER = "answer"          # Final output

@dataclass
class ReasoningStep:
    step_number: int
    step_type: StepType
    description: str
    query: str
    dependencies: List[int]  # Steps that must complete first
    result: Optional[str] = None
    status: Literal["pending", "running", "completed", "failed"] = "pending"

@dataclass
class ReasoningPlan:
    original_query: str
    steps: List[ReasoningStep]
    created_at: float
    
    def get_ready_steps(self) -> List[ReasoningStep]:
        """Get steps ready to execute (dependencies met)."""
        completed = {s.step_number for s in self.steps if s.status == "completed"}
        return [
            s for s in self.steps
            if s.status == "pending"
            and all(d in completed for d in s.dependencies)
        ]
```

## Implementation

### Reasoning Engine

```python
class MultiStepReasoner:
    """
    Execute multi-step reasoning plans.
    
    Time Complexity: O(n * (R + G)) where:
    - n = number of steps
    - R = retrieval cost
    - G = generation cost
    """
    
    def __init__(self, retriever, llm, max_steps: int = 10):
        self.retriever = retriever
        self.llm = llm
        self.max_steps = max_steps
        self.execution_trace: List[Dict] = []
    
    def reason(self, query: str) -> Dict:
        """
        Execute multi-step reasoning on a query.
        
        Returns execution trace and final answer.
        """
        # Step 1: Decompose query into plan
        plan = self.create_plan(query)
        
        # Step 2: Execute plan
        for step_num in range(self.max_steps):
            ready_steps = plan.get_ready_steps()
            
            if not ready_steps:
                if all(s.status == "completed" for s in plan.steps):
                    break  # All done
                else:
                    raise ValueError("Deadlock: steps waiting on each other")
            
            # Execute ready steps (can parallelize independent steps)
            for step in ready_steps:
                result = self.execute_step(step, plan)
                step.result = result
                step.status = "completed"
                
                self.execution_trace.append({
                    "step": step.step_number,
                    "type": step.step_type.value,
                    "query": step.query,
                    "result": result
                })
        
        # Step 3: Return final result
        final_step = [s for s in plan.steps if s.step_type == StepType.ANSWER]
        if final_step:
            return {
                "answer": final_step[0].result,
                "plan": plan,
                "trace": self.execution_trace
            }
        
        return {
            "answer": "Could not complete reasoning",
            "plan": plan,
            "trace": self.execution_trace
        }
    
    def create_plan(self, query: str) -> ReasoningPlan:
        """Create a reasoning plan for the query."""
        prompt = f"""Break this complex query into steps. For each step, specify:
1. Step number
2. Type: RETRIEVE (get info), REASON (analyze), CALCULATE (compute), SYNTHESIZE (combine), ANSWER (final)
3. Description
4. Dependencies (which steps must complete first)

Query: {query}

Format:
1. RETRIEVE | Get X information | dependencies: none
2. REASON | Analyze X | dependencies: 1
..."""
        
        response = self.llm.generate(prompt, max_tokens=500)
        
        # Parse response into steps
        steps = self._parse_plan(response)
        
        return ReasoningPlan(
            original_query=query,
            steps=steps,
            created_at=time.time()
        )
    
    def execute_step(self, step: ReasoningStep, plan: ReasoningPlan) -> str:
        """Execute a single reasoning step."""
        step.status = "running"
        
        if step.step_type == StepType.RETRIEVE:
            return self._execute_retrieve(step)
        
        elif step.step_type == StepType.REASON:
            return self._execute_reason(step, plan)
        
        elif step.step_type == StepType.CALCULATE:
            return self._execute_calculate(step, plan)
        
        elif step.step_type == StepType.SYNTHESIZE:
            return self._execute_synthesize(step, plan)
        
        elif step.step_type == StepType.ANSWER:
            return self._execute_answer(step, plan)
        
        else:
            raise ValueError(f"Unknown step type: {step.step_type}")
    
    def _execute_retrieve(self, step: ReasoningStep) -> str:
        """Execute a retrieval step."""
        docs = self.retriever.search(step.query, k=5)
        return json.dumps([{"id": d["id"], "text": d["text"][:500]} for d in docs])
    
    def _execute_reason(self, step: ReasoningStep, plan: ReasoningPlan) -> str:
        """Execute a reasoning step."""
        # Get dependency results
        context = self._get_dependency_results(step, plan)
        
        prompt = f"""Based on the following information:
{context}

Answer this question: {step.query}

Provide a clear, concise answer."""
        
        return self.llm.generate(prompt)
    
    def _execute_calculate(self, step: ReasoningStep, plan: ReasoningPlan) -> str:
        """Execute a calculation step."""
        # Get dependency results
        context = self._get_dependency_results(step, plan)
        
        prompt = f"""Based on the following information:
{context}

Perform this calculation: {step.query}

Show your work and provide the final result."""
        
        return self.llm.generate(prompt)
    
    def _execute_synthesize(self, step: ReasoningStep, plan: ReasoningPlan) -> str:
        """Execute a synthesis step."""
        # Get all dependency results
        results = []
        for dep_num in step.dependencies:
            dep_step = next(s for s in plan.steps if s.step_number == dep_num)
            results.append(f"Result {dep_num}: {dep_step.result}")
        
        prompt = f"""Synthesize the following results:
{chr(10).join(results)}

Into a coherent response for: {step.query}"""
        
        return self.llm.generate(prompt)
    
    def _execute_answer(self, step: ReasoningStep, plan: ReasoningPlan) -> str:
        """Execute final answer step."""
        # Get all previous results
        context = self._get_all_results(plan)
        
        prompt = f"""Original question: {plan.original_query}

Based on all the work done:
{context}

Provide the final answer."""
        
        return self.llm.generate(prompt)
    
    def _get_dependency_results(self, step: ReasoningStep, plan: ReasoningPlan) -> str:
        """Get results from dependency steps."""
        results = []
        for dep_num in step.dependencies:
            dep_step = next(s for s in plan.steps if s.step_number == dep_num)
            results.append(f"Step {dep_num} result: {dep_step.result}")
        return "\n\n".join(results)
    
    def _get_all_results(self, plan: ReasoningPlan) -> str:
        """Get all completed step results."""
        results = []
        for step in plan.steps:
            if step.status == "completed" and step.result:
                results.append(f"Step {step.step_number} ({step.step_type.value}): {step.result[:200]}...")
        return "\n\n".join(results)
```

## Error Handling

```python
class RobustMultiStepReasoner(MultiStepReasoner):
    """Multi-step reasoner with error recovery."""
    
    def execute_step(self, step: ReasoningStep, plan: ReasoningPlan) -> str:
        """Execute step with error handling."""
        try:
            return super().execute_step(step, plan)
        except Exception as e:
            # Log error
            self.execution_trace.append({
                "step": step.step_number,
                "error": str(e),
                "status": "failed"
            })
            
            # Try recovery strategies
            if step.step_type == StepType.RETRIEVE:
                # Try broader search
                return self._retry_retrieve(step)
            
            elif step.step_type == StepType.CALCULATE:
                # Ask LLM to handle
                return self._fallback_calculate(step, plan)
            
            else:
                # Mark as failed
                step.status = "failed"
                return f"Error: {str(e)}"
    
    def _retry_retrieve(self, step: ReasoningStep) -> str:
        """Retry retrieval with broader query."""
        broader_query = step.query.split()[0:3]  # Simplified
        docs = self.retriever.search(" ".join(broader_query), k=10)
        return json.dumps([{"id": d["id"], "text": d["text"][:500]} for d in docs])
```
