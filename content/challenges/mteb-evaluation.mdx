## The Challenge

Build an MTEBEvaluator class that can evaluate embeddings across:

- **Retrieval tasks**: nDCG, Recall, MRR
- **STS tasks**: Spearman and Pearson correlation
- **Clustering tasks**: V-measure, homogeneity, completeness

## Key Concepts

### Multi-Task Evaluation

```python
class MTEBEvaluator:
    def __init__(self, embed_fn):
        self.embed_fn = embed_fn

    def evaluate_retrieval(self, queries, corpus, qrels, k) -> Dict
    def evaluate_sts(self, s1, s2, scores) -> Dict
    def evaluate_clustering(self, texts, labels, n_clusters) -> Dict
```

### Semantic Textual Similarity (STS)

Compare embedding similarity with human judgments using correlation metrics.

| Metric | Range | Interpretation |
|--------|-------|----------------|
| Spearman ρ | [-1, 1] | Rank correlation |
| Pearson r | [-1, 1] | Linear correlation |

### Clustering Evaluation

Use K-means on embeddings and compare to ground truth labels with V-measure.

| Metric | Description |
|--------|-------------|
| V-measure | Harmonic mean of homogeneity and completeness |
| Homogeneity | Clusters contain single class samples |
| Completeness | Class samples assigned to same cluster |

## Complexity Analysis

| Task | Time Complexity | Space Complexity |
|------|-----------------|------------------|
| Retrieval | O(Q × D × d) | O(D × d) |
| STS | O(N × d) | O(N × d) |
| Clustering | O(N × k × i × d) | O(N × d) |

Where Q = queries, D = documents, N = samples, d = embedding dim, k = clusters, i = iterations

## Implementation

```python
from typing import Dict, List, Tuple
import numpy as np
from scipy.stats import spearmanr, pearsonr
from sklearn.cluster import KMeans
from sklearn.metrics import v_measure_score

class MTEBEvaluator:
    """Evaluate embeddings on MTEB tasks."""
    
    def __init__(self, embed_fn):
        """
        Initialize evaluator with embedding function.
        
        Args:
            embed_fn: Function(text) -> embedding vector
        """
        self.embed_fn = embed_fn
    
    def evaluate_retrieval(
        self,
        queries: Dict[str, str],
        corpus: Dict[str, str],
        qrels: Dict[str, Dict[str, int]],
        k: int = 10
    ) -> Dict[str, float]:
        """
        Evaluate on retrieval task.
        
        Returns:
            Dict with ndcg@k, map@k, recall@k, precision@k
        """
        # Embed all documents
        doc_embeddings = {
            doc_id: self.embed_fn(text) 
            for doc_id, text in corpus.items()
        }
        
        results = {"ndcg": [], "map": [], "recall": [], "precision": []}
        
        for query_id, query_text in queries.items():
            query_emb = self.embed_fn(query_text)
            
            # Compute similarities
            scores = []
            for doc_id, doc_emb in doc_embeddings.items():
                sim = self._cosine_sim(query_emb, doc_emb)
                scores.append((doc_id, sim))
            
            # Sort by similarity
            scores.sort(key=lambda x: x[1], reverse=True)
            retrieved = [doc_id for doc_id, _ in scores[:k]]
            
            # Get ground truth
            relevant = qrels.get(query_id, {})
            relevant_set = {d for d, r in relevant.items() if r > 0}
            
            # Compute metrics
            results["recall"].append(self._recall_at_k(retrieved, relevant_set, k))
            results["precision"].append(self._precision_at_k(retrieved, relevant_set, k))
        
        return {metric: np.mean(scores) for metric, scores in results.items()}
    
    def evaluate_sts(
        self,
        sentences1: List[str],
        sentences2: List[str],
        scores: List[float]
    ) -> Dict[str, float]:
        """
        Evaluate on Semantic Textual Similarity task.
        
        Args:
            sentences1: First set of sentences
            sentences2: Second set of sentences
            scores: Human similarity judgments (0-5)
        
        Returns:
            Dict with spearman and pearson correlations
        """
        # Compute embedding similarities
        pred_scores = []
        for s1, s2 in zip(sentences1, sentences2):
            emb1 = self.embed_fn(s1)
            emb2 = self.embed_fn(s2)
            sim = self._cosine_sim(emb1, emb2)
            pred_scores.append(sim)
        
        # Compute correlations
        spearman, _ = spearmanr(scores, pred_scores)
        pearson, _ = pearsonr(scores, pred_scores)
        
        return {
            "spearman": spearman,
            "pearson": pearson
        }
    
    def evaluate_clustering(
        self,
        texts: List[str],
        labels: List[int],
        n_clusters: int = None
    ) -> Dict[str, float]:
        """
        Evaluate on clustering task.
        
        Args:
            texts: Input texts
            labels: Ground truth cluster labels
            n_clusters: Number of clusters (default: unique labels)
        
        Returns:
            Dict with v_measure, homogeneity, completeness
        """
        if n_clusters is None:
            n_clusters = len(set(labels))
        
        # Embed texts
        embeddings = np.array([self.embed_fn(text) for text in texts])
        
        # Cluster
        kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
        predictions = kmeans.fit_predict(embeddings)
        
        # Compute metrics
        from sklearn.metrics import homogeneity_score, completeness_score
        
        return {
            "v_measure": v_measure_score(labels, predictions),
            "homogeneity": homogeneity_score(labels, predictions),
            "completeness": completeness_score(labels, predictions)
        }
    
    def _cosine_sim(self, a: np.ndarray, b: np.ndarray) -> float:
        """Compute cosine similarity."""
        return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))
    
    def _recall_at_k(self, retrieved: List[str], relevant: set, k: int) -> float:
        """Compute Recall@K."""
        if not relevant:
            return 1.0
        top_k = set(retrieved[:k])
        return len(top_k & relevant) / len(relevant)
    
    def _precision_at_k(self, retrieved: List[str], relevant: set, k: int) -> float:
        """Compute Precision@K."""
        if not retrieved:
            return 0.0
        top_k = retrieved[:k]
        return sum(1 for d in top_k if d in relevant) / len(top_k)
```

from scipy.stats import spearmanr, pearsonr
from sklearn.cluster import KMeans
from sklearn.metrics import v_measure_score, homogeneity_score, completeness_score

class MTEBEvaluator:
    """Evaluate embeddings on MTEB tasks."""
    
    def __init__(self, embed_fn):
        self.embed_fn = embed_fn
    
    def evaluate_retrieval(
        self,
        queries: Dict[str, str],
        corpus: Dict[str, str],
        qrels: Dict[str, Dict[str, int]],
        k: int = 10
    ) -> Dict[str, float]:
        """Evaluate on retrieval task."""
        # Embed corpus
        doc_embeddings = {
            doc_id: self.embed_fn(text) 
            for doc_id, text in corpus.items()
        }
        
        recall_scores = []
        precision_scores = []
        
        for query_id, query_text in queries.items():
            query_emb = self.embed_fn(query_text)
            
            # Compute similarities
            scores = []
            for doc_id, doc_emb in doc_embeddings.items():
                sim = self._cosine_sim(query_emb, doc_emb)
                scores.append((doc_id, sim))
            
            # Sort and get top-k
            scores.sort(key=lambda x: x[1], reverse=True)
            retrieved = [doc_id for doc_id, _ in scores[:k]]
            
            # Get ground truth
            relevant = qrels.get(query_id, {})
            relevant_set = {d for d, r in relevant.items() if r > 0}
            
            # Compute metrics
            recall_scores.append(self._recall_at_k(retrieved, relevant_set, k))
            precision_scores.append(self._precision_at_k(retrieved, relevant_set, k))
        
        return {
            f"recall@{k}": np.mean(recall_scores),
            f"precision@{k}": np.mean(precision_scores)
        }
    
    def evaluate_sts(
        self,
        sentences1: List[str],
        sentences2: List[str],
        scores: List[float]
    ) -> Dict[str, float]:
        """Evaluate on STS task."""
        pred_scores = []
        for s1, s2 in zip(sentences1, sentences2):
            emb1 = self.embed_fn(s1)
            emb2 = self.embed_fn(s2)
            sim = self._cosine_sim(emb1, emb2)
            pred_scores.append(sim)
        
        spearman, _ = spearmanr(scores, pred_scores)
        pearson, _ = pearsonr(scores, pred_scores)
        
        return {"spearman": spearman, "pearson": pearson}
    
    def evaluate_clustering(
        self,
        texts: List[str],
        labels: List[int],
        n_clusters: int = None
    ) -> Dict[str, float]:
        """Evaluate on clustering task."""
        if n_clusters is None:
            n_clusters = len(set(labels))
        
        embeddings = np.array([self.embed_fn(text) for text in texts])
        
        kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
        predictions = kmeans.fit_predict(embeddings)
        
        return {
            "v_measure": v_measure_score(labels, predictions),
            "homogeneity": homogeneity_score(labels, predictions),
            "completeness": completeness_score(labels, predictions)
        }
    
    def _cosine_sim(self, a: np.ndarray, b: np.ndarray) -> float:
        return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-8)

        if not relevant:
            return 1.0
        return len(set(retrieved[:k]) & relevant) / len(relevant)

        if not retrieved:
            return 0.0
        return sum(1 for d in retrieved[:k] if d in relevant) / len(retrieved[:k])
