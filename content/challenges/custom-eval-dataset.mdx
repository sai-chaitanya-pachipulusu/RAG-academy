## Why This Matters

Generic benchmarks (BEIR, MTEB) don't capture your domain's specific challenges. Production RAG systems need custom evaluation suites.

## The Challenge

Build an `EvalDatasetBuilder` class that:

1. Adds documents to a corpus
2. Creates query-relevance examples
3. Validates dataset integrity
4. Exports to JSON format

## Key Concepts

### Builder Pattern

```python
builder = EvalDatasetBuilder("legal_rag_eval", "legal")
builder.add_document("doc1", "Contract law governs...")
builder.add_example(
    query="What happens if someone breaks a contract?",
    relevant_doc_ids=["doc1"],
    difficulty="easy",
    tags=["contract"]
)
dataset = builder.build()
```

### Dataset Validation

- All referenced doc_ids must exist
- Each example must have at least one relevant doc
- No duplicate queries

### Statistics Computation

Track difficulty distribution, tag frequencies, and coverage metrics.

## Complexity Analysis

| Operation | Time Complexity | Space Complexity |
|-----------|-----------------|------------------|
| Add document | O(1) | O(D) |
| Add example | O(R) | O(R) |
| Validate | O(D + R) | O(D) |
| Export | O(D + R) | O(D + R) |

Where D = documents, R = relevance judgments

## Implementation

```python
from typing import List, Dict, Set, Optional
from dataclasses import dataclass, asdict
import json
from collections import defaultdict

@dataclass
class EvalExample:
    """Single evaluation example."""
    query: str
    relevant_doc_ids: List[str]
    difficulty: str  # "easy", "medium", "hard"
    tags: List[str]
    metadata: Optional[Dict] = None

@dataclass
class EvalDataset:
    """Complete evaluation dataset."""
    name: str
    domain: str
    documents: Dict[str, str]  # doc_id -> text
    examples: List[EvalExample]
    
    def to_json(self) -> str:
        """Export to JSON string."""
        data = {
            "name": self.name,
            "domain": self.domain,
            "documents": self.documents,
            "examples": [
                {
                    "query": ex.query,
                    "relevant_doc_ids": ex.relevant_doc_ids,
                    "difficulty": ex.difficulty,
                    "tags": ex.tags,
                    "metadata": ex.metadata
                }
                for ex in self.examples
            ]
        }
        return json.dumps(data, indent=2)
    
    @classmethod
    def from_json(cls, json_str: str) -> "EvalDataset":
        """Load from JSON string."""
        data = json.loads(json_str)
        examples = [
            EvalExample(**ex) for ex in data["examples"]
        ]
        return cls(
            name=data["name"],
            domain=data["domain"],
            documents=data["documents"],
            examples=examples
        )
    
    def get_stats(self) -> Dict:
        """Compute dataset statistics."""
        stats = {
            "num_documents": len(self.documents),
            "num_examples": len(self.examples),
            "difficulty_distribution": defaultdict(int),
            "tag_frequency": defaultdict(int),
            "avg_relevant_docs": 0.0
        }
        
        total_relevant = 0
        for ex in self.examples:
            stats["difficulty_distribution"][ex.difficulty] += 1
            total_relevant += len(ex.relevant_doc_ids)
            for tag in ex.tags:
                stats["tag_frequency"][tag] += 1
        
        if self.examples:
            stats["avg_relevant_docs"] = total_relevant / len(self.examples)
        
        return dict(stats)


class EvalDatasetBuilder:
    """Builder for creating evaluation datasets."""
    
    def __init__(self, name: str, domain: str):
        self.name = name
        self.domain = domain
        self.documents: Dict[str, str] = {}
        self.examples: List[EvalExample] = []
    
    def add_document(self, doc_id: str, text: str) -> "EvalDatasetBuilder":
        """Add a document to the corpus."""
        self.documents[doc_id] = text
        return self
    
    def add_example(
        self,
        query: str,
        relevant_doc_ids: List[str],
        difficulty: str = "medium",
        tags: List[str] = None,
        metadata: Dict = None
    ) -> "EvalDatasetBuilder":
        """Add an evaluation example."""
        if tags is None:
            tags = []
        
        example = EvalExample(
            query=query,
            relevant_doc_ids=relevant_doc_ids,
            difficulty=difficulty,
            tags=tags,
            metadata=metadata
        )
        self.examples.append(example)
        return self
    
    def validate(self) -> List[str]:
        """Validate dataset integrity. Returns list of errors."""
        errors = []
        
        # Check all referenced documents exist
        for i, ex in enumerate(self.examples):
            for doc_id in ex.relevant_doc_ids:
                if doc_id not in self.documents:
                    errors.append(
                        f"Example {i}: Referenced doc '{doc_id}' not found"
                    )
        
        # Check each example has at least one relevant doc
        for i, ex in enumerate(self.examples):
            if not ex.relevant_doc_ids:
                errors.append(f"Example {i}: No relevant documents specified")
        
        # Check for duplicate queries
        queries = [ex.query for ex in self.examples]
        if len(queries) != len(set(queries)):
            errors.append("Duplicate queries found")
        
        return errors
    
    def build(self) -> EvalDataset:
        """Build and return the evaluation dataset."""
        errors = self.validate()
        if errors:
            raise ValueError(f"Validation failed: {errors}")
        
        return EvalDataset(
            name=self.name,
            domain=self.domain,
            documents=self.documents,
            examples=self.examples
        )
```

## Best Practices

1. **Stratified sampling**: Ensure balanced difficulty distribution
2. **Diverse queries**: Cover different query types (factoid, list, comparison)
3. **Multiple annotators**: Reduce bias with consensus labeling
4. **Version control**: Track dataset versions like code
5. **Regular updates**: Refresh with new documents and queries
